\title{The Hilbert series of the focused arc algebra}
\author{Diego Salazar}
\date{\today}
\maketitle

All rings considered are commutative with identity element $1$, and all ring homomorphisms preserve the identity.
Let $A$ be a ring.
An \defterm{$A$-algebra} is a ring $B$ equipped with a ring homomorphism $A \to B$.
An \defterm{$\mathbb{N}$-graded ring} is a ring $R$ together with a family $(R_n)_{n \in \mathbb{N}}$ of additive subgroups of $R$ such that $R = \bigoplus_{n \in \mathbb{N}} R_n$ and $R_mR_n \subseteq R_{m + n}$.
Let $k$ be a field.
An \defterm{$\mathbb{N}$-graded $k$-algebra} is a $k$-algebra $A = \bigoplus_{n \in \mathbb{N}}A_n$ such that $A$ is an $\mathbb{N}$-graded ring, and each additive subgroup $A_n$ is a subspace of $A$.
If $a \in A_n$, we say $a$ has \defterm{weight} $n$.
This automatically implies $1 \in A_0$ and $k \subseteq A_0$.

Let $A$ be an $\mathbb{N}$-graded $k$-algebra.
The \defterm{Hilbert-Poincaré series of $A$} is defined by
\begin{equation*}
  HP_A(t) = \sum_{n \in \mathbb{N}}\dim_k(A_n)t^n.
\end{equation*}
If each $A_n$ is finite dimensional, then $HP_A(t) \in \mathbb{N}[[t]]$.

A \defterm{differential $k$-algebra} is a $k$-algebra $A$ together with a $k$-linear derivation $\partial\in Der_k(A)$.
An \defterm{$\mathbb{N}$-graded differential $k$-algebra} is a differential $k$-algebra $A$ that is also an $\mathbb{N}$-graded $k$-algebra satisfying $\partial(A_n) \subseteq A_{n + 1}$.

Let $k$ be a field and let $A$ be a $k$-algebra.
Let $JA$ be the jet algebra of $A$.
We have a natural monomorphism $A \hookrightarrow JA$, so we can consider $A$ as a subset of $JA$ with $k[A]_\partial = JA$, where the notation $k[S]_\partial$ indicates the differential $k$-subalgebra generated by a subset $S$.
We define
\begin{equation*}
  (JA)_n = span_k\{(\partial^{n_1}a_1)\dots(\partial^{n_s}a_s) \mid s, n_1, \dots, n_s \in \mathbb{N}, a_1, \dots, a_s \in A, n_1 + \dots + n_s = n\}.
\end{equation*}
Then
\begin{equation*}
  JA = \bigoplus_{n \in \mathbb{N}}(JA)_n,
\end{equation*}
and $JA$ becomes an $\mathbb{N}$-graded differential $k$-algebra.
We note that $(JA)_0 = A$.

Let $\mathfrak{p} \in Spec(A)$.
Then $A_\mathfrak{p}$ is a local ring with maximal ideal $\mathfrak{p}A_\mathfrak{p}$, and the quotient $\kappa(\mathfrak{p}) = A_\mathfrak{p}/\mathfrak{p}A_\mathfrak{p}$ is the \defterm{residue field of $A$ at $\mathfrak{p}$}.
We have a natural homomorphism $A \to \kappa(\mathfrak{p})$ which makes $\kappa(\mathfrak{p})$ into an $A$-algebra.
We can also view $JA$ as an $A$-algebra. 
The \defterm{focused arc algebra of $A$ at $\mathfrak{p}$} is the $\mathbb{N}$-graded $\kappa(\mathfrak{p})$-algebra
\begin{equation}
  \label{eq:1}
  J^\mathfrak{p}A = JA \otimes_A \kappa(\mathfrak{p})
\end{equation}
with operation given by
\begin{equation*}
  (a\otimes x)(b\otimes y) = (ab)\otimes(xy)
\end{equation*}
and with $\mathbb{N}$-grading given by
\begin{equation*}
(J^\mathfrak{p}A)_n = (JA)_n \otimes_A \kappa(\mathfrak{p}).
\end{equation*}
We note that $\dim_{\kappa(\mathfrak{p})}((J^\mathfrak{p}A)_0) = 1$ because $(J^\mathfrak{p}A)_0 = A \otimes_A \kappa(\mathfrak{p}) = \kappa(\mathfrak{p})$, and $J^\mathfrak{p}A$ does not inherit a derivation from $JA$.

We are interested in the case where
\begin{equation}
  \label{eq:2}
  A = k[X_1, \dots, X_n]/(F_1, \dots, F_m),
\end{equation}
where $F_1, \dots, F_m \in k[X_1, \dots, X_n]$ satisfy
\begin{equation}
  \label{eq:3}
  0 = F_r(0, \dots, 0), \qquad r = 1, \dots, m,
\end{equation}
and $\mathfrak{p} = (X_1 + (F_1, \dots, F_m), \dots, X_n + (F_1, \dots, F_m))$ is a maximal ideal.
In this case, the residue field is $\kappa(\mathfrak{p}) = k$, and the natural homomorphism $A \to k$ is just evaluation at $(0, \dots, 0)$.
Informally, we can think of $JA$ as being the following set of solutions
\begin{equation*}
  JA = \{(x_1(t), \dots, x_n(t)) \in k[[t]]^n \mid 0 = F_r(x_1(t), \dots, x_n(t)), r = 1, \dots, m\}
\end{equation*}
in the same way we can think of $A$ as
\begin{equation*}
  A = \{(x_1, \dots, x_n) \in k^n \mid 0 = F_r(x_1, \dots, x_n), r = 1, \dots, m\}.
\end{equation*}
Of course, this is just an intuition because the $k$-algebra $A$ may be nonreduced.

More precisely, we consider the polynomial $k$-algebra $k[X^{(i)}_j \mid i \in \mathbb{N}, j = 1, \dots, n]$ in countably many variables $X^{(i)}_j$ with derivation $\partial \in Der_k(k[X^{(i)}_j \mid i \in \mathbb{N}, j = 1, \dots, n])$ given by
\begin{equation}
  \label{eq:4}
  \partial X^{(i)}_j = X^{(i + 1)}_j, \qquad i \in \mathbb{N}, j = 1, \dots, n.
\end{equation}
Identifying $X_j = X^{(0)}_j, j = 1, \dots, n$, we have $k[X_1, \dots, X_n]_\partial = k[X^{(i)}_j \mid i \in \mathbb{N}, j = 1, \dots, n]$.

Following our intuition, we construct the jet algebra $JA$ by imposing the relations $F_r = 0$.
We define the sequence of higher derivatives $F_r^{(l)}$ by successively applying the derivation $\partial$. That is, we define $F_r^{(0)} = F_r(X^{(0)}_1, \dots, X^{(0)}_n)$ and set
\begin{equation*}
  F_r^{(l)} = \partial^l(F_r^{(0)}), \qquad l \in \mathbb{N}, r = 1, \dots, m.
\end{equation*}
By definition, the differential ideal generated by $(F_1, \dots, F_m)$ is
\begin{equation*}
  (F_1, \dots, F_m)_\partial = (F^{(l)}_r \mid l \in \mathbb{N}, r = 1, \dots, m).
\end{equation*}
We then set
\begin{equation}
  \label{eq:8}
  JA = \frac{k[X^{(i)}_j \mid i \in \mathbb{N}, j = 1, \dots, n]}{(F_1, \dots, F_m)_\partial},
\end{equation}
and we have a natural inclusion $A \hookrightarrow JA$ of $k$-algebras satisfying our desired universal property.
We note that we haven't assumed $char(k) = 0$ in this construction of the jet algebra.

Alternatively, the graded structure on $JA$ can be defined more intrinsically via the natural action of the extension fields $K$ of $k$ on the arc parameter $t$.
Here, we again assume $char(k)$ might not be $0$.
Let $k \hookrightarrow K$ be a field extension of $k$.
For any $\lambda \in K^\times$, we wish to define a $K$-algebra automorphism $\Phi_\lambda \in Aut_K(K[X^{(i)}_j \mid i \in \mathbb{N}, j = 1, \dots, n])$ by setting its action on the generators:
\begin{equation}
  \label{eq:11}
  \Phi_\lambda(X^{(i)}_j) = \lambda^iX^{(i)}_j.
\end{equation}
To motivate this definition, we consider the formal arcs
\begin{equation*}
  X_j(t) = \sum_{i = 0}^\infty X^{(i)}_j t^i \in K[X^{(i)}_j \mid i \in \mathbb{N}, j = 1, \dots, n][[t]], \qquad j = 1, \dots, n.
\end{equation*}
We wish to have
\begin{equation}
  \label{eq:12}
  \Phi_\lambda(F(X_1(t), \dots, X_n(t))) = F(X_1(\lambda t), \dots, X_n(\lambda t)), \qquad F \in K[X_1, \dots, X_n].
\end{equation}
Indeed, taking $F = X_j, j = 1, \dots, n$ perfectly realizes this because
\begin{equation*}
  X_j(\lambda t) = \sum_{i = 0}^\infty (X^{(i)}_j\lambda^i)t^i,
\end{equation*}
and with this definition, we get \eqref{eq:12} for $F \in K[X_1, \dots, X_n]$.
Moreover, $G \in k[X^{(i)}_j \mid i \in \mathbb{N}, j = 1, \dots, n]$ has weight $d$ if and only if
\begin{equation*}
  \Phi_\lambda(G\otimes1) = G\otimes\lambda^d
\end{equation*}
for all field extensions $k \hookrightarrow K$ and all $\lambda \in K^\times$.
It is necessary to consider field extensions $k \hookrightarrow K$ because if, for example, $k = \mathbb{Z}/2\mathbb{Z}$, then $\lambda^d = \lambda$ for all $d \in \mathbb{N}$ and all $\lambda \in k^\times$.

To ensure $\Phi_\lambda$ is well-defined on the quotient $JA$, we must check that it preserves the defining ideal.
For the base polynomials $F_r \in k[X_1, \dots, X_n]$, the variables $X_j = X^{(0)}_j$ have weight $0$, so $\Phi_\lambda(F_r) = F_r$.
The derivation $\partial$ interacts with this automorphism via the commutation relation $\Phi_\lambda \circ \partial = \lambda \partial \circ \Phi_\lambda$.
Using the general characteristic-free definition of $\partial$ from \eqref{eq:4}, we can prove this explicitly on the generators:
\begin{align*}
  \Phi_\lambda(\partial X^{(i)}_j) &= \Phi_\lambda(X^{(i+1)}_j) = \lambda^{i+1}X^{(i+1)}_j, \\
  \lambda \partial(\Phi_\lambda(X^{(i)}_j)) &= \lambda \partial(\lambda^i X^{(i)}_j) = \lambda \cdot \lambda^i X^{(i+1)}_j = \lambda^{i+1}X^{(i+1)}_j.
\end{align*}
Note that this commutation relation also holds for the characteristic zero definition \eqref{eq:4}, as the $(i+1)$ factor is simply carried through as a scalar.
Because $\Phi_\lambda$ is a $k$-algebra automorphism and $\partial$ is a $k$-linear derivation, this relation extends from the generators to the entire polynomial ring.
Applying this relation recursively to $F^{(l)}_r = \partial^l F_r$ yields
\begin{equation*}
  \Phi_\lambda(F^{(l)}_r\otimes1) = F^{(l)}_r\otimes\lambda^l.
\end{equation*}
Since the generators of the differential ideal are simply scaled by $\lambda^l$, the ideal $(F_1, \dots, F_m)_\partial\otimes K$ is mapped entirely into itself.
Thus, $\Phi_\lambda$ descends to a well-defined automorphism $\Phi_\lambda: JA \otimes_k K \to JA \otimes_k K$. 
Therefore, an element $f \in JA$ is homogeneous of weight $d$ if and only if
\begin{equation*}
  \Phi_\lambda(f\otimes1) = f\otimes\lambda^d
\end{equation*}
for all field extensions $k \hookrightarrow K$ and all $\lambda \in K^\times$.

Now we move onto abstract schemes.
Again, we consider $k$ a field, and let $X$ be a $k$-scheme of finite type.
Here, we again assume $char(k)$ might not be $0$.
For any $k$-algebra $K$ and any scalar $\lambda \in K^\times$, we have an automorphism of $K$-algebras
\begin{align*}
  \varphi_\lambda:K[[t]] &\to K[[t]], \\
  t &\mapsto \lambda t.
\end{align*}
This automorphism induces a natural scheme morphism
\begin{equation*}
  Spec(\varphi_\lambda): Spec(K[[t]]) \to Spec(K[[t]]),
\end{equation*}
which yields a natural bijection
\begin{equation*}
  \circ Spec(\varphi_\lambda): Hom_k(Spec(K[[t]]), X) \to Hom_k(Spec(K[[t]]), X).
\end{equation*}
Since $Hom_k(Spec(K[[t]]), X) = Hom_k(Spec(K), JX) = JX(K)$, for $\lambda \in K^\times$, we have a bijection
\begin{equation*}
  \Phi_\lambda: JX(K) \to JX(K),
\end{equation*}
which simply corresponds to mapping a formal arc $x(t) \in JX(K)$ to $x(\lambda t)$.
Because $JX(K)$ is merely a set, we cannot directly apply algebraic functors to it.
To recover the rigorous algebraic automorphism defined earlier, we must lift this geometric action back into the category of schemes.
We do this in four steps.

\subsubsection*{1. Base change}
Because we want to evaluate everything over the field extension $K$, we mathematically lift the underlying scheme $X$ to $K$ via the fiber product $X_K = X \times_k Spec(K)$.

\subsubsection*{2. Jet schemes commute with base change}
A fundamental and powerful property of jet schemes is that their formation commutes with base change.
This means that the jet scheme of the base-changed scheme $X_K$ is naturally isomorphic to the base change of the original jet scheme: $J_K(X_K) \cong (JX)_K$.
Because of this isomorphism, the set of points $JX(K)$ can naturally be viewed as the $K$-valued points of the $K$-scheme $J_K(X_K)$.

\subsubsection*{3. Inducing the scheme automorphism}
Since $J_K(X_K)$ represents a functor of points on all $K$-algebras, its $S$-valued points for any $K$-algebra $S$ are $Hom_{K-Sch}(Spec(S[[t]]), X_K)$.
The automorphism $\varphi_\lambda: K[[t]] \to K[[t]]$ extends canonically to an $S$-algebra automorphism mapping $t \mapsto \lambda t$.
This induces a natural transformation on the functor of points via pre-composition $\gamma \mapsto \gamma \circ Spec(\varphi_{\lambda, S})$.
By Yoneda's Lemma, this natural transformation corresponds to a unique formal $K$-scheme automorphism
\begin{equation*}
  \tilde{\Phi}_\lambda: (JX)_K \to (JX)_K
\end{equation*}
on the base-changed scheme itself.

\subsubsection*{4. Applying the global sections functor}
The global sections operator $\Gamma$ is a contravariant functor from the category of schemes to the category of commutative rings.
Applying $\Gamma$ to the $K$-scheme automorphism $\tilde{\Phi}_\lambda$ formally yields a ring automorphism $\Gamma(\tilde{\Phi}_\lambda): \Gamma((JX)_K) \to \Gamma((JX)_K)$ on its coordinate ring.
Substituting the commuting identity $\Gamma((JX)_K) = \Gamma(JX) \otimes_k K$, we perfectly recover the algebraic automorphism
\begin{equation*}
  \Gamma(\tilde{\Phi}_\lambda): \Gamma(JX) \otimes_k K \to \Gamma(JX) \otimes_k K.
\end{equation*}
We define an element $f \in \Gamma(JX)$ to be algebraically homogeneous of weight $d$ if
\begin{equation*}
  \Gamma(\tilde{\Phi}_\lambda)(f\otimes1) = f\otimes\lambda^d
\end{equation*}
for all field extensions $k \hookrightarrow K$ and all $\lambda \in K^\times$.
By defining this homogeneity entirely over field extensions, we bypass the need for arbitrary test rings while still robustly protecting against finite-field collapses.

\subsection*{The universal scaling homomorphism $k[[t]] \to k[\lambda][[t]]$}

Alternatively, instead of evaluating across an infinite family of field extensions $k \hookrightarrow K$,
we can capture the scaling action intrinsically and universally in a single test ring:
the polynomial ring $k[\lambda]$ representing the affine line $\mathbb{A}^1_k$.

Consider the continuous $k$-algebra homomorphism
\begin{align*}
  \psi: k[[t]] &\to k[\lambda][[t]], \\
  t &\mapsto \lambda t.
\end{align*}
On any formal power series,
this morphism simply scales the parameter by the formal indeterminate $\lambda$:
\begin{equation*}
  \psi\left(\sum_{i = 0}^\infty a_i t^i\right) = \sum_{i = 0}^\infty a_i (\lambda t)^i = \sum_{i = 0}^\infty (a_i \lambda^i) t^i \in k[\lambda][[t]].
\end{equation*}

Taking the spectrum yields an induced morphism on the formal disks:
\begin{equation*}
  Spec(\psi): Spec(k[\lambda][[t]]) \to Spec(k[[t]]).
\end{equation*}
For any $k$-scheme $X$,
pre-composition with $Spec(\psi)$ maps any $k$-arc $\gamma \in \operatorname{Hom}_k(Spec(k[[t]]), X) = JX(k)$
to an arc defined over the test ring $k[\lambda]$:
\begin{equation*}
  \gamma \circ Spec(\psi) \in \operatorname{Hom}_k(Spec(k[\lambda][[t]]), X) = JX(k[\lambda]).
\end{equation*}
By Yoneda's Lemma,
this universal assignment across all $k$-algebras defines an honest morphism in the category of schemes:
\begin{equation*}
  \mu: \mathbb{A}^1_k \times JX \to JX,
\end{equation*}
which corresponds to the algebraic action of the multiplicative monoid $\mathbb{A}^1_k$ on the arc scheme.

Applying the global sections functor $\Gamma$,
the comorphism $\mu^\sharp$ of this action yields a $k$-algebra homomorphism directly on the coordinate rings:
\begin{equation*}
  \sigma: JA \to JA \otimes_k k[\lambda] \cong JA[\lambda].
\end{equation*}
To determine the action of $\sigma$ on the coordinate generators,
we evaluate $\psi$ on the canonical formal arc $X_j(t) = \sum_{i = 0}^\infty X^{(i)}_j t^i \in JA[[t]]$:
\begin{equation*}
  X_j(\lambda t) = \sum_{i = 0}^\infty X^{(i)}_j (\lambda t)^i = \sum_{i = 0}^\infty (\lambda^i X^{(i)}_j) t^i \in JA[\lambda][[t]].
\end{equation*}
Reading off the coefficient of $t^i$,
we see that $\sigma$ acts on the generators by
\begin{equation*}
  \sigma(X^{(i)}_j) = \lambda^i X^{(i)}_j.
\end{equation*}
An element $f \in JA$ is then defined to be homogeneous of weight $d$ if and only if
\begin{equation*}
  \sigma(f) = \lambda^d f \in JA[\lambda].
\end{equation*}

Because $\lambda$ is a formal indeterminate in $k[\lambda]$,
the powers $\lambda^d$ are linearly independent over $JA$.
This single algebraic condition works uniformly over any base field $k$ (finite or infinite, and in arbitrary characteristic)
without requiring testing across field extensions,
while remaining entirely intrinsic and coordinate-free.
