\section{The Hilbert--Poincaré series and the focussed arc algebra}
\label{sec:hilb-poinc-seri}

All rings considered are commutative with identity element $1$, and all ring homomorphisms preserve the identity.
Let $A$ be a ring.
An \defterm{$A$-algebra} is a ring $B$ equipped with a ring homomorphism $A \to B$.
An \defterm{$\mathbb{N}$-graded ring} is a ring $R$ together with a family $(R_w)_{w \in \mathbb{N}}$ of additive subgroups of $R$ such that $R = \bigoplus_{w \in \mathbb{N}} R_w$ and $R_{w_1}R_{w_2} \subseteq R_{w_1 + w_2}$.
Let $k$ be a field.
An \defterm{$\mathbb{N}$-graded $k$-algebra} is a $k$-algebra $A = \bigoplus_{w \in \mathbb{N}}A_w$ such that $A$ is an $\mathbb{N}$-graded ring, and each additive subgroup $A_w$ is a subspace of $A$.
If $a \in A_w$, we say $a$ has \defterm{weight} $w$ and write $\operatorname{wt}(a) = w$.
This automatically implies $1 \in A_0$ and $k \subseteq A_0$.

Let $A$ be an $\mathbb{N}$-graded $k$-algebra.
The \defterm{Hilbert--Poincaré series of $A$} is defined by
\begin{equation*}
  \operatorname{HP}_A(t) = \sum_{w \in \mathbb{N}}\dim_k(A_w)t^w.
\end{equation*}
If each $A_w$ is finite dimensional, then $\operatorname{HP}_A(t) \in \mathbb{N}[[t]]$.

A \defterm{differential $k$-algebra} is a $k$-algebra $A$ together with a $k$-linear derivation $\partial \in \operatorname{Der}_k(A)$.
An ideal $I \subseteq A$ is \defterm{differential} if $\partial(I) \subseteq I$.
An \defterm{$\mathbb{N}$-graded differential $k$-algebra} is a differential $k$-algebra $A$ that is also an $\mathbb{N}$-graded $k$-algebra satisfying $\partial(A_w) \subseteq A_{w + 1}$.

Let $k$ be a field and let $A$ be a $k$-algebra.
Let $J_\infty A$ be the arc algebra of $A$.
We have a natural monomorphism $A \hookrightarrow J_\infty A$, so we can consider $A$ as a subset of $J_\infty A$ with $k[A]_\partial = J_\infty A$, where the notation $k[S]_\partial$ indicates the differential $k$-subalgebra generated by a subset $S$.
We define
\begin{equation*}
  (J_\infty A)_w = \operatorname{span}_k\{(\partial^{w_1}a_1)\dots(\partial^{w_s}a_s) \mid s, w_1, \dots, w_s \in \mathbb{N}, a_1, \dots, a_s \in A, w_1 + \dots + w_s = w\}.
\end{equation*}
Then
\begin{equation*}
  J_\infty A = \bigoplus_{w \in \mathbb{N}}(J_\infty A)_w,
\end{equation*}
and $J_\infty A$ becomes an $\mathbb{N}$-graded differential $k$-algebra.
We note that $(J_\infty A)_0 = A$.

Let $\mathfrak{p} \in \operatorname{Spec}(A)$.
Then $A_\mathfrak{p}$ is a local ring with maximal ideal $\mathfrak{p}A_\mathfrak{p}$, and the quotient $\kappa(\mathfrak{p}) = A_\mathfrak{p}/\mathfrak{p}A_\mathfrak{p}$ is the \defterm{residue field of $A$ at $\mathfrak{p}$}.
We have a natural homomorphism $A \to \kappa(\mathfrak{p})$ which makes $\kappa(\mathfrak{p})$ into an $A$-algebra.
We can also view $J_\infty A$ as an $A$-algebra.
The \defterm{focussed arc algebra of $A$ at $\mathfrak{p}$} is the $\mathbb{N}$-graded $\kappa(\mathfrak{p})$-algebra
\begin{equation}
  \label{eq:1}
  J_\infty^\mathfrak{p} A = J_\infty A \otimes_A \kappa(\mathfrak{p})
\end{equation}
with operation given by
\begin{equation*}
  (a\otimes x)(b\otimes y) = (ab)\otimes(xy)
\end{equation*}
and with $\mathbb{N}$-grading given by
\begin{equation*}
  (J_\infty^\mathfrak{p} A)_w = (J_\infty A)_w \otimes_A \kappa(\mathfrak{p}) = \{a \otimes x \mid a \in (J_\infty A)_w, x \in \kappa(\mathfrak{p})\}.
\end{equation*}
We note that $\dim_{\kappa(\mathfrak{p})}((J_\infty^\mathfrak{p} A)_0) = 1$ because $(J_\infty^\mathfrak{p} A)_0 = A \otimes_A \kappa(\mathfrak{p}) = \kappa(\mathfrak{p})$, and $J_\infty^\mathfrak{p} A$ does not inherit a derivation from $J_\infty A$.

\section{Concrete examples of the focussed arc algebra}
\label{sec:concr-exampl-focus}

We are interested in the case where
\begin{equation}
  \label{eq:2}
  A = k[X_1, \dots, X_n]/(F_1, \dots, F_m),
\end{equation}
where $F_1, \dots, F_m \in k[X_1, \dots, X_n]$ satisfy
\begin{equation}
  \label{eq:3}
  0 = F_r(0, \dots, 0), \qquad r = 1, \dots, m,
\end{equation}
and $\mathfrak{p} = (X_1 + (F_1, \dots, F_m), \dots, X_n + (F_1, \dots, F_m))$ is a maximal ideal.
In this case, the residue field is $\kappa(\mathfrak{p}) = k$, and the natural homomorphism $A \to k$ is just evaluation at $(0, \dots, 0)$.
Informally, we can think of $J_\infty A$ as being the following set of solutions
\begin{equation*}
  J_\infty A = \{(x_1(t), \dots, x_n(t)) \in k[[t]]^n \mid 0 = F_r(x_1(t), \dots, x_n(t)), r = 1, \dots, m\}
\end{equation*}
in the same way we can think of $A$ as
\begin{equation*}
  A = \{(x_1, \dots, x_n) \in k^n \mid 0 = F_r(x_1, \dots, x_n), r = 1, \dots, m\}.
\end{equation*}
Of course, this is just an intuition because the $k$-algebra $A$ may be nonreduced.
For example, if $A = k[X]/(X^2)$ (a case we are interested in), then with this definition, we would get $A = J_\infty A = \{0\}$, which we obviously don't want.
We give two ways of constructing $J_\infty A$ and then show an isomorphism between them.

Using formal arcs: We consider the polynomial $k$-algebra $k[X^{(i)}_j \mid i = 0, 1, \dots, j = 1, \dots, n]$ in countably many variables $X^{(i)}_j$ with derivation $\partial \in \operatorname{Der}_k(k[X^{(i)}_j \mid i = 0, 1, \dots, j = 1, \dots, n])$ given by
\begin{equation}
  \label{eq:4}
  \partial X^{(i)}_j = (i + 1)X^{(i + 1)}_j, \qquad i = 0, 1, \dots, j = 1, \dots, n,
\end{equation}
where we assume $\operatorname{char}(k) = 0$.

To motivate this definition, we define:
\begin{equation}
  \label{eq:5}
  X_j(t) = X_j^{(0)} + X_j^{(1)}t + \dots \in k[X^{(i)}_j \mid i = 0, 1, \dots, j = 1, \dots, n][[t]], \qquad j = 1, \dots, n.
\end{equation}
We have a natural derivation $\partial_t$.
We wish to have
\begin{equation}
  \label{eq:6}
  \partial(F(X_1(t), \dots, X_n(t))) = \partial_t(F(X_1(t), \dots, X_n(t))), \qquad F \in k[X_1, \dots, X_n].
\end{equation}
Taking $F = X_j, j = 1, \dots, n$, we get \eqref{eq:4}.
With this definition, we get \eqref{eq:6} for $F \in k[X_1, \dots, X_n]$.
Identifying $X_j = X^{(0)}_j, j = 1, \dots, n$, we have $k[X_1, \dots, X_n]_\partial = k[X^{(i)}_j \mid i = 0, 1, \dots, j = 1, \dots, n]$.

Following our intuition, we write
\begin{equation}
  \label{eq:7}
  F_r(X_1(t), \dots, X_n(t)) = F_r^{(0)} + F_r^{(1)}t + \dots, \qquad r = 1, \dots, m,
\end{equation}
where $F_r^{(l)} \in k[X^{(i)}_j \mid i = 0, 1, \dots, j = 1, \dots, n], l \in \mathbb{N}, r = 1, \dots, m$.
Then $(F^{(l)}_r \mid l \in \mathbb{N}, r = 1, \dots, m)$ is a differential ideal because $\partial(F_r(X_1(t), \dots, X_n(t))) = \partial_t(F_r(X_1(t), \dots, X_n(t))), r = 1, \dots, m$ implies
\begin{equation*}
  \partial(F_r^{(l)}) = (l + 1)F_r^{(l + 1)}, \qquad l \in \mathbb{N}, r = 1, \dots, m.
\end{equation*}
Thus
\begin{equation*}
  F_r^{(l)} = \frac{\partial^lF_r^{(0)}}{l!}, \qquad l \in \mathbb{N}, r = 1, \dots, m.
\end{equation*}
From \eqref{eq:7}, we see that $F_r^{(0)} = F_r, r = 1, \dots, m$.
We have proved that
\begin{equation*}
  (F^{(l)}_r \mid l \in \mathbb{N}, r = 1, \dots, m) = (F_1, \dots, F_m)_\partial.
\end{equation*}
We set
\begin{equation}
  \label{eq:8}
  J_\infty A = \frac{k[X^{(i)}_j \mid i = 0, 1, \dots, j = 1, \dots, n]}{(F^{(l)}_r \mid l \in \mathbb{N}, r = 1, \dots, m)} = \frac{k[X^{(i)}_j \mid i = 0, 1, \dots, j = 1, \dots, n]}{(F_1, \dots, F_m)_\partial},
\end{equation}
which comes equipped with a natural inclusion $A \hookrightarrow J_\infty A$ of $k$-algebras.

Using derivations: We consider the polynomial $k$-algebra $k[Y^{(i)}_j \mid i = 0, 1, \dots, j = 1, \dots, n]$ in countably many variables $Y^{(i)}_j$ with derivation $\partial \in \operatorname{Der}_k(k[Y^{(i)}_j \mid i = 0, 1, \dots, j = 1, \dots, n])$ given by
\begin{equation}
  \label{eq:10}
  \partial Y^{(i)}_j = Y^{(i + 1)}_j, \qquad i = 0, 1, \dots, j = 1, \dots, n.
\end{equation}
Identifying $Y_j = Y^{(0)}_j, j = 1, \dots, n$, we have $k[Y_1, \dots, Y_n]_\partial = k[Y^{(i)}_j \mid i = 0, 1, \dots, j = 1, \dots, n]$.

Following our intuition, we construct the arc algebra $J_\infty A$ by imposing the relations $F_r = 0$.
We define the sequence of higher derivatives $F_r^{(l)}$ by successively applying the derivation $\partial$.
That is, we define $F_r^{(0)} = F_r(Y^{(0)}_1, \dots, Y^{(0)}_n)$ and set
\begin{equation*}
  F_r^{(l)} = \partial^l(F_r^{(0)}), \qquad l \in \mathbb{N}, r = 1, \dots, m.
\end{equation*}
By definition, the differential ideal generated by $(F_1, \dots, F_m)$ is
\begin{equation*}
  (F_1, \dots, F_m)_\partial = (F^{(l)}_r \mid l \in \mathbb{N}, r = 1, \dots, m).
\end{equation*}
We then set
\begin{equation}
  \label{eq:11}
  J_\infty A = \frac{k[Y^{(i)}_j \mid i = 0, 1, \dots, j = 1, \dots, n]}{(F_1, \dots, F_m)_\partial},
\end{equation}
which likewise comes equipped with a natural inclusion $A \hookrightarrow J_\infty A$ of $k$-algebras.
We note that in the formal power series expansion~\eqref{eq:7},
the coefficient ideal $(F_r^{(l)} \mid l \in \mathbb{N}, \, r = 1, \dots, m)$ is defined over any field $k$ without dividing by $l!$
(corresponding to Hasse--Schmidt higher derivations),
whereas expressing $F_r^{(l)} = \partial^l F_r^{(0)} / l!$ via the derivation~\eqref{eq:4} requires $\operatorname{char}(k) = 0$.
Similarly,
while the ordinary derivation construction~\eqref{eq:11} does not divide by $l!$,
in characteristic $p > 0$ the derivation $\partial$ kills $p$-th powers ($\partial(g^p) = p g^{p - 1}\partial g = 0$;
for example, if $F = Y^p$, then $(Y^p)_\partial = ((Y^{(0)})^p)$, whereas the coefficient ideal of $X(t)^p$ is $((X^{(i)})^p \mid i = 0, 1, \dots)$).
When $\operatorname{char}(k) = 0$,
these two constructions are isomorphic as differential algebras via $Y_j^{(i)} \mapsto i!X^{(i)}_j$ for $i = 0, 1, \dots$ and $j = 1, \dots, n$.

In both ways, setting
\begin{equation*}
  f^{(l)}_r = F_r^{(l)}|_{X^{(0)}_j \mapsto 0, j = 1, \dots, n} \in k[X^{(i)}_j \mid i = 1, 2, \dots, j = 1, \dots, n], \qquad l = 1, 2, \dots, r = 1, \dots, m,
\end{equation*}
we have
\begin{equation}
  \label{eq:9}
  J_\infty^0 A = J_\infty^\mathfrak{p} A = J_\infty A \otimes_A k = \frac{k[X^{(i)}_j \mid i = 1, 2, \dots, j = 1, \dots, n]}{(f^{(l)}_r \mid l = 1, 2, \dots, r = 1, \dots, m)}.
\end{equation}

\section{The grading of the arc algebra}
\label{sec:grading-arc-algebra}

Now we wish to define the grading of an arc algebra $J_\infty A$ where, as in the previous section,
\begin{equation*}
  A = k[X_1, \dots, X_n]/(F_1, \dots, F_m),
\end{equation*}
where $F_1, \dots, F_m \in k[X_1, \dots, X_n]$ satisfy
\begin{equation*}
  0 = F_r(0, \dots, 0), \qquad r = 1, \dots, m.
\end{equation*}

We endow the polynomial $k$-algebra $k[X^{(i)}_j \mid i = 0, 1, \dots, j = 1, \dots, n]$ with the $\mathbb{N}$-grading defined on variables by $\operatorname{wt}(X^{(i)}_j) = i$,
so that a monomial $\prod_{s = 1}^p X^{(i_s)}_{j_s}$ has weight $i_1 + \dots + i_p$.
Equivalently,
letting $\lambda$ be an indeterminate and considering the $k[\lambda]$-algebra homomorphism
\begin{equation*}
  k[\lambda][X^{(i)}_j \mid i = 0, 1, \dots, j = 1, \dots, n] \longrightarrow k[\lambda][X^{(i)}_j \mid i = 0, 1, \dots, j = 1, \dots, n], \qquad
  X^{(i)}_j \longmapsto \lambda^i X^{(i)}_j,
\end{equation*}
a polynomial $F \in k[X^{(i)}_j \mid i = 0, 1, \dots, j = 1, \dots, n]$ is homogeneous of weight $w$ if and only if
\begin{equation*}
  (F)_{X^{(i)}_j \mapsto \lambda^i X^{(i)}_j} = \lambda^w F.
\end{equation*}
Under the substitution $X^{(i)}_j \mapsto \lambda^i X^{(i)}_j$,
the formal power series $X_j(t) = \sum_{i = 0}^\infty X^{(i)}_j t^i$ becomes $X_j(\lambda t)$.
Consequently,
replacing $t$ by $\lambda t$ in~\eqref{eq:7} gives
\begin{equation*}
  \sum_{l \in \mathbb{N}} (F_r^{(l)})_{X^{(i)}_j \mapsto \lambda^i X^{(i)}_j} t^l
  = F_r\left(X_1(\lambda t), \dots, X_n(\lambda t)\right)
  = \sum_{l \in \mathbb{N}} \lambda^l F_r^{(l)} t^l,
\end{equation*}
which shows that each $F_r^{(l)}$ (and likewise its specialization $f_r^{(l)}$ at $X^{(0)}_j = 0$) is homogeneous of weight $l$ for all $l \in \mathbb{N}$ and $r = 1, \dots, m$
(in the derivation construction~\eqref{eq:11}, since $F_1, \dots, F_m$ are homogeneous of weight $0$ and $\partial$ increases weight by $1$, the generators $F_r^{(l)} = \partial^l F_r^{(0)}$ are likewise homogeneous of weight $l$).
Thus $(F_r^{(l)} \mid l \in \mathbb{N}, r = 1, \dots, m)$ and $(f_r^{(l)} \mid l = 1, 2, \dots, r = 1, \dots, m)$ are homogeneous ideals,
inducing well-defined $\mathbb{N}$-gradings on $J_\infty A$ and $J_\infty^0 A$.

\begin{example}
  \label{exa:1}
  For $A = k[X]$, we have
  \begin{equation*}
    \operatorname{HP}_{J_\infty^0 A}(t) = \prod_{i = 1}^\infty \frac{1}{1 - t^i} = \mathbb{H}(t).
  \end{equation*}
  Recall that a \defterm{partition $\lambda = [\lambda_1, \dots, \lambda_r]$ of $m \in \mathbb{N}$} is finite $r$-tuple such that $\lambda_1 \le \lambda_2 \le \dots \le \lambda_r$, and we denote the number of partitions
  The series $\mathbb{H}(t)$ is generating series for partitions:
  \begin{equation*}
    \mathbb{H}(t) = \sum_{n = 0}^\infty p(n)t^n.
  \end{equation*}
\end{example}

\begin{example}
  \label{exa:2}
  For $A = k[X]/(X^2)$ with $\operatorname{char}(k) = 0$, we have
  \begin{equation*}
    \operatorname{HP}_{J_\infty^0 A}(t) = \prod^\infty_{\substack{i = 1 \\ i \not\equiv 0, 1, 4 \bmod 5}} \frac{1}{1 - t^i}.
  \end{equation*}
\end{example}

\section{The Focussed Arc Algebra of the Non-Reduced Node and Rogers--Ramanujan Identities}
\label{sec:focused-arc-algebra-xy2}

Let $k$ be a field of characteristic zero,
let $e \ge 1$ be an integer,
and let $s \in \{1, \dots, e\}$.
For $m \in \mathbb{N}$, let $(t)_m = \prod_{i = 1}^m (1 - t^i)$ denote the $q$-Pochhammer symbol (with $(t)_0 = 1$).
The Andrews--Gordon generalization of the Rogers--Ramanujan identities states that
\begin{equation*}
  \sum_{k = (k_1, \dots, k_{e - 1}) \in \mathbb{N}^{e - 1}}\frac{t^{\frac{1}{2}kG^{(e)}k^\top + kB^{(e)}_{e - s}}}{(t)_{k_1}\dots(t)_{k_{e - 1}}} = \prod^\infty_{\substack{i = 1 \\ i \not\equiv 0, \pm s \bmod 2e + 1}}(1 - t^i)^{-1},
\end{equation*}
where
\begin{align*}
  G^{(e)} &= (2\min\{i, j\})_{i, j = 1}^{e - 1}, \\
  B^{(e)}_j &= (\underbrace{0, \dots, 0}_{e - 1 - j}, 1, 2, \dots, j)^\top \in \mathbb{N}^{e - 1} \quad \text{for $j = 0, 1, \dots, e - 1$},
\end{align*}
and ${}^\top$ denotes the transpose matrix (in particular, $B^{(e)}_0 = (0, \dots, 0)^\top$).
We denote by $AG_{e, s}(t) \in \mathbb{N}[[t]]$ either side of this identity (though it's probably better to think of the products when doing Hilbert--Poincaré series).

Recall that
\begin{equation}
  \mathbb{H}(t) = \prod_{i=1}^\infty \frac{1}{1 - t^i} = \sum_{m=0}^\infty p(m)t^m \in \mathbb{N}[[t]]
\end{equation}
is the generating function for integer partitions.

\begin{conjecture}
\label{conj:xy2-hilbert-series}
For $A_2 = k[X, Y] / ((XY)^2)$, we have
\begin{equation}
  \label{eq:conj-xy2-hp}
  \operatorname{HP}_{J_\infty^0(A_2)}(t) = \frac{\mathbb{H}(t)}{1 - t}AG_{2, 1}(t).
\end{equation}
Equivalently,
in Euler product form,
\begin{equation}
  \label{eq:conj-xy2-euler}
  \operatorname{HP}_{J_\infty^0(A_2)}(t) = \prod_{k=1}^\infty \frac{1}{(1 - t^k)^{c_k}},
\end{equation}
where the exponents $c_k$ satisfy $c_1 = 2$ and,
for all $k \ge 2$,
\begin{equation}
  c_k = \begin{cases}
    2 & \text{if } k \equiv \pm 2 \pmod 5, \\
    1 & \text{if } k \equiv 0, \pm 1 \pmod 5.
  \end{cases}
\end{equation}
\end{conjecture}

\begin{conjecture}[Generalization to $(XY)^e$]
\label{conj:xye-generalization}
Let $e \ge 2$ be an integer.
For the algebra $A_e = k[X, Y]/((XY)^e)$, we have
\begin{equation}
  \operatorname{HP}_{J_\infty^0(A_e)}(t) = \frac{\mathbb{H}(t)}{1 - t}AG_{e, 1}(t).
\end{equation}
\end{conjecture}

\begin{remark}
\label{rmk:status-of-conjectures}
Conjectures~\ref{conj:xy2-hilbert-series} and~\ref{conj:xye-generalization} were originally formulated from numerical Gröbner basis computations.
We keep their statement as conjectures above for historical reference,
and prove them in Corollary~\ref{cor:proof-xye-notes} below as the $n = 2$ specialization of Theorem~\ref{thm:normal-crossings-notes} (and more generally Theorem~\ref{thm:general-regular-germs}).
\end{remark}

\subsection{Finite-Variable Jet Truncation and Graded Freeness}
\label{subsec:finite-jet-truncation}

We now establish Conjecture~\ref{conj:xy2-hilbert-series} and Conjecture~\ref{conj:xye-generalization}
(together with their generalization to normal crossings $(X_1 \cdots X_n)^e = 0$)
using finite-variable jet truncations for the reduced hypersurface $F = 0$,
avoiding both multivariable Gröbner bases and homological $\operatorname{Tor}$ modules.

More generally, let $F \in k[X_1, \dots, X_n]$ be a polynomial vanishing at the origin with order $d = \operatorname{ord}_0(F) \ge 1$, and let $A_{F, e} = k[X_1, \dots, X_n] / (F^e)$ for $e \ge 1$, writing $A_F := A_{F, 1} = k[X_1, \dots, X_n] / (F)$ for the reduced hypersurface (the results of this subsection and Theorem~\ref{thm:normal-crossings-notes}(1) hold over a field $k$ of arbitrary characteristic;
$\operatorname{char}(k) = 0$ is used only in Subsection~\ref{subsec:shifted-fiber}).
Let
\begin{equation}
  P_\infty = k[X_j^{(i)} \mid 1 \le j \le n, \, i \ge 1]
\end{equation}
be the polynomial ring in the positive-weight arc variables endowed with the grading $\operatorname{wt}(X_j^{(i)}) = i$,
whose Hilbert--Poincaré series is $\operatorname{HP}_{P_\infty}(t) = \prod_{i = 1}^\infty (1 - t^i)^{-n} = \mathbb{H}(t)^n$,
and for each truncation order $N \ge d$, let
\begin{equation}
  P_{\le N} = k[X_j^{(i)} \mid 1 \le j \le n, \, 1 \le i \le N] \subseteq P_\infty, \qquad
  \operatorname{HP}_{P_{\le N}}(t) = \prod_{i = 1}^N \frac{1}{(1 - t^i)^n}.
\end{equation}
Writing $X_j(t) = \sum_{i = 1}^\infty X_j^{(i)} t^i \in P_\infty[[t]]$ for $j = 1, \dots, n$,
we define the homogeneous weight-$l$ polynomials
\begin{equation}
  u_l := [t^l] F(X_1(t), \dots, X_n(t)) \in P_\infty, \qquad l \ge d,
\end{equation}
where $[t^l]$ denotes the coefficient of $t^l$
(since $\operatorname{ord}_0(F) = d$, we have $[t^l]F(X_1(t), \dots, X_n(t)) = 0$ for $l < d$;
moreover, for $l \le N$, $u_l$ does not depend on any variable $X_j^{(i)}$ with $i > N$ and hence lies in $P_{\le N}$).

For the reduced hypersurface $A_F = k[X_1, \dots, X_n] / (F)$,
its focussed arc algebra at the origin and its $N$-th focussed jet truncation ($N \ge d$) are
\begin{equation*}
  J_\infty^0(A_F) = \frac{P_\infty}{(u_l \mid l \ge d)}, \qquad
  J_N^0(A_F) = \frac{P_{\le N}}{(u_d, u_{d + 1}, \dots, u_N)}.
\end{equation*}
Because all variables $X_j^{(i)}$ ($i > N$) and all relations $u_l$ ($l > N$) omitted from $J_N^0(A_F)$ are homogeneous of weight $\ge N + 1$,
the graded components of $J_N^0(A_F)$ and $J_\infty^0(A_F)$ coincide up to weight $N$:
\begin{equation}
  \left(J_N^0(A_F)\right)_w = \left(J_\infty^0(A_F)\right)_w \qquad \text{for all } 0 \le w \le N,
\end{equation}
and therefore in the $(t)$-adic topology on $\mathbb{Z}[[t]]$
(cf.\ Proposition~3.2 of~\cite{bruschek_arc_2013}):
\begin{equation}
  \label{eq:hp-t-adic-limit}
  \operatorname{HP}_{J_\infty^0(A_F)}(t) = \lim_{N \to \infty} \operatorname{HP}_{J_N^0(A_F)}(t).
\end{equation}

Next, let $Q_\infty = k[V^{(1)}, V^{(2)}, \dots]$ be the polynomial ring in the arc variables of a single coordinate $V$,
endowed with the \defterm{shifted grading}
\begin{equation}
  \operatorname{wt}_d(V^{(i)}) := i + d - 1 \qquad (i \ge 1),
\end{equation}
whose Hilbert--Poincaré series with respect to $\operatorname{wt}_d$ is
\begin{equation}
  \operatorname{HP}_{Q_\infty}(t) = \prod_{i = 1}^\infty \frac{1}{1 - t^{i + d - 1}} = \prod_{l = d}^\infty \frac{1}{1 - t^l}
\end{equation}
(and for each $N \ge d$, let $Q_{\le N} = k[V^{(1)}, \dots, V^{(N - d + 1)}] \subseteq Q_\infty$ be the subring of variables of $\operatorname{wt}_d$-weight $\le N$,
with $\operatorname{HP}_{Q_{\le N}}(t) = \prod_{l = d}^N (1 - t^l)^{-1}$).
Define the homomorphism of $\mathbb{N}$-graded $k$-algebras
\begin{equation}
  \phi_F \colon Q_\infty \longrightarrow P_\infty, \qquad V^{(i)} \longmapsto u_{i + d - 1} \quad (i \ge 1).
\end{equation}
In $Q_\infty[[t]]$, let $V(t) = \sum_{i = 1}^\infty V^{(i)} t^i$ and consider the focussed arc algebra of the 1-variable $e$-fold point $k[V]/(V^e)$:
\begin{equation}
  J_\infty^0\left(k[V]/(V^e)\right) = \frac{Q_\infty}{I_\infty^0(V^e)}, \qquad
  I_\infty^0(V^e) = \left( f_p(V) \;\middle|\; p \ge e \right), \qquad
  f_p(V) := [t^p] V(t)^e.
\end{equation}
For each $p \ge e$, the Cauchy product expansion gives
\begin{equation*}
  f_p(V) = [t^p] V(t)^e = \sum_{\substack{i_1 + \dots + i_e = p \\ i_1, \dots, i_e \ge 1}} V^{(i_1)} \cdots V^{(i_e)},
\end{equation*}
which is homogeneous of standard weight $p$ and polynomial degree $e$,
and is therefore homogeneous with respect to the shifted grading $\operatorname{wt}_d$ of weight
\begin{equation*}
  \operatorname{wt}_d(f_p) = (i_1 + d - 1) + \dots + (i_e + d - 1) = p + e(d - 1).
\end{equation*}
Thus $I_\infty^0(V^e)$ is a $\operatorname{wt}_d$-homogeneous ideal,
making $J_\infty^0(k[V]/(V^e))$ into an $\mathbb{N}$-graded $k$-algebra under $\operatorname{wt}_d$.
Moreover,
since $\phi_F(t^{d - 1} V(t)) = \sum_{i = 1}^\infty u_{i + d - 1} t^{i + d - 1} = F(X_1(t), \dots, X_n(t))$,
raising both sides to the $e$-th power and extracting the coefficient of $t^l$ for $l = p + e(d - 1) \ge e d$ ($p \ge e$) yields
\begin{equation*}
  \phi_F\left(f_p(V)\right)
  = \sum_{\substack{i_1 + \dots + i_e = p \\ i_1, \dots, i_e \ge 1}} u_{i_1 + d - 1} \cdots u_{i_e + d - 1}
  = [t^{p + e(d - 1)}] \left(F(X_1(t), \dots, X_n(t))\right)^e,
\end{equation*}
so the defining ideal $I_\infty^0(F^e) = \left([t^l](F(X_1(t), \dots, X_n(t)))^e \mid l \ge e d\right)$ of $J_\infty^0(A_{F, e})$ satisfies $I_\infty^0(F^e) = \phi_F(I_\infty^0(V^e)) P_\infty$.
Viewing $P_\infty$ as an $\mathbb{N}$-graded $Q_\infty$-algebra via $\phi_F$,
we obtain a canonical isomorphism of $\mathbb{N}$-graded $k$-algebras
\begin{equation}
  \label{eq:JN-quotient-phi}
  J_\infty^0(A_{F, e}) = \frac{P_\infty}{\phi_F(I_\infty^0(V^e)) P_\infty} \cong P_\infty \otimes_{Q_\infty} \frac{Q_\infty}{I_\infty^0(V^e)} = P_\infty \otimes_{Q_\infty} J_\infty^0\left(k[V]/(V^e)\right).
\end{equation}

\begin{proposition}[Elementary Graded Freeness and Factored Hilbert Series]
\label{prop:graded-freeness-finite}
Suppose that for every truncation order $N \ge d$,
$(u_d, u_{d + 1}, \dots, u_N)$ is a regular sequence in $P_{\le N}$.
Then:
\begin{enumerate}
  \item For each $N \ge d$, the Hilbert--Poincaré series of $J_N^0(A_F) = P_{\le N} / (u_d, \dots, u_N)$ is
  \begin{equation}
    \label{eq:HP-RN-quotient}
    \operatorname{HP}_{J_N^0(A_F)}(t) = \operatorname{HP}_{P_{\le N}}(t) \prod_{l = d}^N (1 - t^l) = \frac{\operatorname{HP}_{P_{\le N}}(t)}{\operatorname{HP}_{Q_{\le N}}(t)},
  \end{equation}
  and taking the limit as $N \to \infty$ gives the Hilbert--Poincaré series of $J_\infty^0(A_F)$:
  \begin{equation}
    \label{eq:HP-RF-quotient}
    \operatorname{HP}_{J_\infty^0(A_F)}(t) = \frac{\operatorname{HP}_{P_\infty}(t)}{\operatorname{HP}_{Q_\infty}(t)} = \frac{\mathbb{H}(t)^{n - 1}}{(t)_{d - 1}}.
  \end{equation}
  \item Viewing $P_\infty$ as a graded $Q_\infty$-module via $\phi_F$,
  choose any homogeneous $k$-vector space basis $(e_\alpha)_{\alpha \in \mathcal{I}}$ of $J_\infty^0(A_F)$,
  and lift each $e_\alpha$ to a homogeneous element $\tilde{e}_\alpha \in P_\infty$ of the same weight $w_\alpha = \operatorname{wt}(e_\alpha)$.
  Then the graded $Q_\infty$-linear map
  \begin{equation}
    \label{eq:Phi-free-iso}
    \Phi \colon \bigoplus_{\alpha \in \mathcal{I}} Q_\infty(-w_\alpha) \longrightarrow P_\infty, \qquad
    (g_\alpha(V))_{\alpha \in \mathcal{I}} \longmapsto \sum_{\alpha \in \mathcal{I}} \phi_F(g_\alpha(V)) \tilde{e}_\alpha
  \end{equation}
  is an isomorphism of graded $Q_\infty$-modules
  (equivalently, $P_\infty \cong J_\infty^0(A_F) \otimes_k Q_\infty$ as a graded $Q_\infty$-module by~\eqref{eq:app-shifted-sum-tensor};
  see Appendix~\ref{app:graded-tensor-products} for the grading conventions on shifted modules and tensor products).
  \item For every integer $e \ge 1$,
  tensoring~\eqref{eq:Phi-free-iso} with $- \otimes_{Q_\infty} J_\infty^0(k[V]/(V^e))$ induces an isomorphism of $\operatorname{wt}_d$-graded $J_\infty^0(k[V]/(V^e))$-modules (and of $\mathbb{N}$-graded $k$-vector spaces)
  \begin{equation}
    \label{eq:JN-direct-sum-BnN}
    J_\infty^0(A_{F, e}) \cong P_\infty \otimes_{Q_\infty} J_\infty^0\left(k[V]/(V^e)\right) \cong \bigoplus_{\alpha \in \mathcal{I}} J_\infty^0\left(k[V]/(V^e)\right)(-w_\alpha) \cong J_\infty^0(A_F) \otimes_k J_\infty^0\left(k[V]/(V^e)\right),
  \end{equation}
  and consequently,
  by~\eqref{eq:app-hp-tensor-product},
  \begin{equation}
    \label{eq:HP-JN-factored}
    \operatorname{HP}_{J_\infty^0(A_{F, e})}(t) = \operatorname{HP}_{J_\infty^0(A_F)}(t) \cdot \operatorname{HP}_{\left(J_\infty^0(k[V]/(V^e)), \, \operatorname{wt}_d\right)}(t) = \frac{\mathbb{H}(t)^{n - 1}}{(t)_{d - 1}} \cdot \operatorname{HP}_{\left(J_\infty^0(k[V]/(V^e)), \, \operatorname{wt}_d\right)}(t).
  \end{equation}
\end{enumerate}
\end{proposition}

\begin{proof}
(1) Fix $N \ge d$.
For each $l = d, \dots, N$,
set $M_{l - 1} := P_{\le N} / (u_d, \dots, u_{l - 1})$ (with $M_{d - 1} := P_{\le N}$).
Since $u_l$ is a homogeneous non-zero-divisor of weight $l$ on $M_{l - 1}$,
the short exact sequence
\begin{equation*}
  0 \longrightarrow M_{l - 1}(-l) \xrightarrow{\cdot u_l} M_{l - 1} \longrightarrow M_{l - 1} / u_l M_{l - 1} \longrightarrow 0
\end{equation*}
together with~\eqref{eq:app-hp-shift} gives $\operatorname{HP}_{M_l}(t) = (1 - t^l) \operatorname{HP}_{M_{l - 1}}(t)$.
Applying this $N - d + 1$ times yields~\eqref{eq:HP-RN-quotient}:
\begin{equation*}
  \operatorname{HP}_{J_N^0(A_F)}(t) = \frac{\prod_{i = 1}^N (1 - t^i)^{-n}}{\prod_{l = d}^N (1 - t^l)^{-1}} = \frac{\prod_{i = 1}^N (1 - t^i)^{-(n - 1)}}{(1 - t)\cdots(1 - t^{d - 1})}.
\end{equation*}
Taking the limit as $N \to \infty$ in the $(t)$-adic topology via~\eqref{eq:hp-t-adic-limit} gives~\eqref{eq:HP-RF-quotient}.

(2) We first show that $\Phi$ is surjective by induction on the weight $w \ge 0$.
For $w = 0$,
$(P_\infty)_0 = k$ is spanned by the lift of $1 \in J_\infty^0(A_F)$.
For $w \ge 1$ and any $x \in (P_\infty)_w$,
since $(e_\alpha)_{w_\alpha = w}$ spans $(J_\infty^0(A_F))_w$,
there exists a $k$-linear combination $y = \sum_{w_\alpha = w} c_\alpha \tilde{e}_\alpha \in \operatorname{Im}(\Phi)$ such that $x - y \in (u_l \mid l \ge d)_w$.
Because $\operatorname{wt}(u_l) = l$,
only generators $u_l$ with $d \le l \le w$ can contribute in weight $w$,
so we can write
\begin{equation*}
  x - y = \sum_{l = d}^w u_l z_{w - l}
\end{equation*}
with $z_{w - l} \in (P_\infty)_{w - l}$.
Since $w - l < w$,
each $z_{w - l}$ lies in $\operatorname{Im}(\Phi)$ by the induction hypothesis;
since $\operatorname{Im}(\Phi)$ is a $Q_\infty$-submodule of $P_\infty$,
each $u_l z_{w - l} = \phi_F(V^{(l - d + 1)}) z_{w - l}$ also lies in $\operatorname{Im}(\Phi)$,
proving $x \in \operatorname{Im}(\Phi)$.

On the other hand,
by~\eqref{eq:app-hp-shift} and~\eqref{eq:HP-RF-quotient},
the Hilbert--Poincaré series of the domain of $\Phi$ is
\begin{equation*}
  \operatorname{HP}_{\bigoplus_{\alpha \in \mathcal{I}} Q_\infty(-w_\alpha)}(t)
  = \operatorname{HP}_{Q_\infty}(t) \sum_{\alpha \in \mathcal{I}} t^{w_\alpha}
  = \operatorname{HP}_{Q_\infty}(t) \cdot \operatorname{HP}_{J_\infty^0(A_F)}(t)
  = \operatorname{HP}_{P_\infty}(t).
\end{equation*}
Hence,
in every weight $w \ge 0$,
the restriction $\Phi_w$ is a surjective $k$-linear map between two finite-dimensional $k$-vector spaces of the same dimension,
and is therefore an isomorphism.

(3) Tensoring the graded $Q_\infty$-module isomorphism~\eqref{eq:Phi-free-iso} with $- \otimes_{Q_\infty} J_\infty^0(k[V]/(V^e))$ and using~\eqref{eq:JN-quotient-phi},~\eqref{eq:app-shifted-base-change}, and~\eqref{eq:app-shifted-sum-tensor} gives
\begin{equation*}
  J_\infty^0(A_{F, e}) \cong P_\infty \otimes_{Q_\infty} J_\infty^0\left(k[V]/(V^e)\right)
  \cong \bigoplus_{\alpha \in \mathcal{I}} \left(Q_\infty(-w_\alpha) \otimes_{Q_\infty} J_\infty^0\left(k[V]/(V^e)\right)\right)
  \cong \bigoplus_{\alpha \in \mathcal{I}} J_\infty^0\left(k[V]/(V^e)\right)(-w_\alpha)
  \cong J_\infty^0(A_F) \otimes_k J_\infty^0\left(k[V]/(V^e)\right),
\end{equation*}
and taking Hilbert--Poincaré series via~\eqref{eq:app-hp-tensor-product} and~\eqref{eq:HP-RF-quotient} immediately yields~\eqref{eq:HP-JN-factored}.
\end{proof}

\subsection{The Shifted Grading on $J_\infty^0(k[V]/(V^e))$}
\label{subsec:shifted-fiber}

We now determine $\operatorname{HP}_{\left(J_\infty^0(k[V]/(V^e)), \, \operatorname{wt}_d\right)}(t)$ for the 1-variable fat-point focussed arc algebra $J_\infty^0(k[V]/(V^e)) = Q_\infty / (f_p(V) \mid p \ge e)$ endowed with the shifted grading $\operatorname{wt}_d(V^{(i)}) = i + d - 1$
(here $\operatorname{char}(k) = 0$ is used).

In Section~5 of Bruschek, Mourtada, and Schepers~\cite{bruschek_arc_2013},
the defining polynomials of $J_\infty^0(k[y]/(y^e))$ in the derivation model $k[y_1, y_2, \dots]$ are $f_p^{\mathrm{BMS}}(y) = \bigl.D^p(y_0^e)\bigr|_{y_0 = 0} = e!\,B_{p, e}(y_1, \dots, y_{p - e + 1})$.
Since $\operatorname{char}(k) = 0$,
the diagonal rescaling $y_i = i!\,V^{(i)}$ satisfies $f_p^{\mathrm{BMS}}(y) = p!\,f_p(V)$.
By Lemma~5.3 of~\cite{bruschek_arc_2013},
$(f_p)_{p \ge e}$ is a Gröbner basis of $I_\infty^0(V^e)$ under the weighted reverse lexicographic order,
with leading monomials
\begin{equation}
  \label{eq:shifted-leading-monomials}
  \operatorname{lm}(f_{qe + j}) = (V^{(q)})^{e - j} (V^{(q + 1)})^j \qquad (q \ge 1, \; 0 \le j \le e - 1).
\end{equation}
Consequently,
the standard monomials (those not divisible by any $\operatorname{lm}(f_{qe + j})$),
\begin{equation*}
  V^{(\lambda_1)} \cdots V^{(\lambda_\ell)} \qquad (\lambda_1 \ge \dots \ge \lambda_\ell \ge 1, \; \lambda_i - \lambda_{i + e - 1} \ge 2),
\end{equation*}
form a $k$-vector space basis of $J_\infty^0(k[V]/(V^e))$.
Under the shifted grading $\operatorname{wt}_d(V^{(i)}) = i + d - 1$,
each factor $V^{(\lambda_r)}$ has weight $\mu_r := \lambda_r + d - 1 \ge d$,
so the basis monomial $V^{(\lambda_1)} \cdots V^{(\lambda_\ell)}$ is $\operatorname{wt}_d$-homogeneous of weight $|\mu| = \mu_1 + \dots + \mu_\ell$.
Re-indexing by the shifted partitions $\mu = (\mu_1, \dots, \mu_\ell)$ (which satisfy $\mu_i - \mu_{i + e - 1} = \lambda_i - \lambda_{i + e - 1} \ge 2$ and $\mu_{\mathrm{min}} \ge d$),
this basis restricts to a $k$-basis of each $\operatorname{wt}_d$-homogeneous component of $J_\infty^0(k[V]/(V^e))$,
yielding
\begin{equation}
  \label{eq:tau-HP-BnN}
  \operatorname{HP}_{\left(J_\infty^0(k[V]/(V^e)), \, \operatorname{wt}_d\right)}(t) = \sum_{\substack{\mu \\ \mu_i - \mu_{i + e - 1} \ge 2 \\ \mu_{\mathrm{min}} \ge d}} t^{|\mu|}.
\end{equation}

\subsection{Powers of Regular Hypersurface Germs and Proof of the Conjectures}
\label{subsec:normal-crossings-proof}
\label{subsec:regular-hypersurface-germs}

We now combine the graded freeness isomorphism of Proposition~\ref{prop:graded-freeness-finite}
with the shifted partition series of $J_\infty^0(k[V]/(V^e))$ in~\eqref{eq:tau-HP-BnN}
to compute the Hilbert--Poincaré series of powers of any hypersurface germ whose truncated jet polynomials form a regular sequence.

\begin{theorem}[Hilbert--Poincaré Series for Powers of Regular Hypersurface Germs]
\label{thm:general-regular-germs}
Let $n \ge 1$ and $e \ge 1$ be integers.
Let $F \in k[X_1, \dots, X_n]$ be a polynomial vanishing at the origin with order of vanishing $d = \operatorname{ord}_0(F) \ge 1$,
and let $A_{F, e} = k[X_1, \dots, X_n] / (F^e)$.
Suppose that for every truncation order $N \ge d$,
the truncated jet polynomials $(u_d, u_{d + 1}, \dots, u_N)$ form a regular sequence in $P_{\le N}$.
Then:
\begin{enumerate}
  \item The Hilbert--Poincaré series of the focussed arc algebra $J_\infty^0(A_{F, e})$ is
  \begin{equation}
    \label{eq:HP-general-regular}
    \operatorname{HP}_{J_\infty^0(A_{F, e})}(t) = \frac{\mathbb{H}(t)^{n - 1}}{(t)_{d - 1}} \sum_{\substack{\mu \\ \mu_i - \mu_{i + e - 1} \ge 2 \\ \mu_{\mathrm{min}} \ge d}} t^{|\mu|},
  \end{equation}
  where the summation runs over all integer partitions $\mu = (\mu_1 \ge \mu_2 \ge \dots)$ satisfying the Gordon gap condition $\mu_i - \mu_{i + e - 1} \ge 2$ and smallest part $\mu_{\mathrm{min}} \ge d$,
  including the empty partition $\mu = \emptyset$ with $|\emptyset| = 0$.
  \item For $e = 1$ (the reduced hypersurface $A_F = k[X_1, \dots, X_n] / (F)$),
  the gap condition $\mu_i - \mu_i \ge 2$ cannot be satisfied by any partition with at least one part,
  so the summation collapses to the single term $t^0 = 1$ for $\mu = \emptyset$,
  yielding
  \begin{equation}
    \label{eq:HP-general-n1}
    \operatorname{HP}_{J_\infty^0(A_F)}(t) = \frac{\mathbb{H}(t)^{n - 1}}{(t)_{d - 1}}.
  \end{equation}
  \item For $e = 2$,
  the summation is over partitions with difference $\mu_i - \mu_{i + 1} \ge 2$ and $\mu_{\mathrm{min}} \ge d$,
  yielding the shifted Rogers--Ramanujan series:
  \begin{equation}
    \label{eq:HP-general-n2}
    \operatorname{HP}_{J_\infty^0(k[X_1, \dots, X_n] / (F^2))}(t) = \frac{\mathbb{H}(t)^{n - 1}}{(t)_{d - 1}} \sum_{\ell = 0}^\infty \frac{t^{\ell^2 + (d - 1)\ell}}{(t)_\ell}.
  \end{equation}
  \item For $n = 1$ and $F(X) = X^d$,
  the prefactor is $1 / (t)_{d - 1}$,
  and for $d = 1$ ($F(X) = X$, the $e$-fold point),
  the prefactor collapses to $1$,
  recovering Theorem~5.6 of Bruschek, Mourtada, and Schepers~\cite{bruschek_arc_2013}.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Substituting~\eqref{eq:tau-HP-BnN} into~\eqref{eq:HP-JN-factored} of Proposition~\ref{prop:graded-freeness-finite}(3) immediately yields~\eqref{eq:HP-general-regular}.

(2) When $e = 1$,
a partition with $\ell \ge 1$ parts would have to satisfy $\mu_1 - \mu_1 \ge 2$,
which is $0 \ge 2$,
an impossibility.
Thus no non-empty partition satisfies the condition,
and only the empty partition $\mu = \emptyset$ (having $0$ parts and $|\emptyset| = 0$) contributes,
giving the sum $t^0 = 1$ and establishing~\eqref{eq:HP-general-n1}.

(3) For $e = 2$,
a partition $\mu = (\mu_1 > \mu_2 > \dots > \mu_\ell)$ of length $\ell \ge 0$ satisfies $\mu_i - \mu_{i + 1} \ge 2$ and $\mu_\ell \ge d$ if and only if $\mu_i \ge d + 2(\ell - i)$ for all $1 \le i \le \ell$.
The minimal such partition has size
\begin{equation*}
  \sum_{i = 1}^\ell \left(d + 2(\ell - i)\right) = \ell d + 2 \frac{(\ell - 1)\ell}{2} = \ell^2 + (d - 1)\ell.
\end{equation*}
Setting $\nu_i := \mu_i - \left(d + 2(\ell - i)\right)$,
the sequence $\nu = (\nu_1 \ge \nu_2 \ge \dots \ge \nu_\ell \ge 0)$ is an arbitrary partition with at most $\ell$ parts,
whose generating function is well-known to be $1 / (t)_\ell$ (cf.\ Theorem~1.1 of~\cite{andrews_partitions_1976}).
Summing over all lengths $\ell \ge 0$ yields~\eqref{eq:HP-general-n2}.

(4) Setting $n = 1$ and $F(X) = X^d$ gives $\mathbb{H}(t)^{n - 1} = \mathbb{H}(t)^0 = 1$,
so the prefactor is $1 / (t)_{d - 1}$.
When $d = 1$,
$(t)_{d - 1} = (t)_0 = 1$,
so the prefactor is $1$,
recovering Theorem~5.6 of~\cite{bruschek_arc_2013}.
\end{proof}

We now specialize Theorem~\ref{thm:general-regular-germs} to the normal crossings singularities $F(X_1, \dots, X_n) = X_1 \cdots X_n$,
which have order of vanishing $d = n$.

\begin{theorem}[Hilbert--Poincaré Series of $(X_1 \cdots X_n)^e = 0$]
\label{thm:normal-crossings-notes}
Let $n \ge 1$ and $e \ge 1$ be integers,
and let $A_{n, e} = k[X_1, \dots, X_n] / ((X_1 \cdots X_n)^e)$.
Then:
\begin{enumerate}
  \item For every truncation order $N \ge n$,
  the polynomials $u_l = [t^l](X_1(t) \cdots X_n(t))$ for $n \le l \le N$ form a regular sequence $(u_n, u_{n + 1}, \dots, u_N)$ in $P_{\le N}$
  (cf.\ Proposition~3.9 of~\cite{bruschek_arc_2013},
  citing Theorem~2.2 of Goward and Smith~\cite{goward_smith_2006}).
  \item The Hilbert--Poincaré series of the focussed arc algebra $J_\infty^0(A_{n, e})$ is
  \begin{equation}
    \label{eq:HP-normal-crossings}
    \operatorname{HP}_{J_\infty^0(A_{n, e})}(t) = \frac{\mathbb{H}(t)^{n - 1}}{(t)_{n - 1}} \sum_{\substack{\mu \\ \mu_i - \mu_{i + e - 1} \ge 2 \\ \mu_{\mathrm{min}} \ge n}} t^{|\mu|}.
  \end{equation}
  \item For $e = 1$,
  the series collapses to
  \begin{equation}
    \label{eq:HP-normal-crossings-n1}
    \operatorname{HP}_{J_\infty^0(k[X_1, \dots, X_n] / (X_1 \cdots X_n))}(t) = \frac{\mathbb{H}(t)^{n - 1}}{(t)_{n - 1}},
  \end{equation}
  recovering Proposition~3.9 of Bruschek, Mourtada, and Schepers~\cite{bruschek_arc_2013}.
  \item In particular,
  for $e = 2$,
  we obtain the shifted Rogers--Ramanujan series:
  \begin{equation}
    \label{eq:HP-normal-crossings-n2}
    \operatorname{HP}_{J_\infty^0(A_{n, 2})}(t) = \frac{\mathbb{H}(t)^{n - 1}}{(t)_{n - 1}} \sum_{\ell = 0}^\infty \frac{t^{\ell^2 + (n - 1)\ell}}{(t)_\ell}.
  \end{equation}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Fix $N \ge n$.
(As noted in Proposition~3.9 of~\cite{bruschek_arc_2013},
the fact that $(u_n, \dots, u_N)$ forms a regular sequence follows from the equidimensionality
of the jet schemes of normal crossings varieties established by Goward and Smith~\cite{goward_smith_2006};
below we give a direct self-contained proof via prime ideals in $P_{\le N}$.)
Let $\mathfrak{q} \subseteq P_{\le N}$ be any prime ideal containing $I := (u_n, \dots, u_N)$,
and let $D := P_{\le N} / \mathfrak{q}$ be the quotient integral domain.
In $D[t]$,
write $\overline{X_j}(t) = \sum_{i = 1}^N \overline{X_j^{(i)}} t^i$ for the image of $X_j(t)$.
By definition of $u_l = [t^l](X_1(t) \cdots X_n(t))$,
the inclusion $I \subseteq \mathfrak{q}$ is equivalent to
\begin{equation*}
  \overline{X_1}(t) \cdots \overline{X_n}(t) \equiv 0 \pmod{t^{N + 1}} \qquad \text{in } D[t].
\end{equation*}
Because $D$ is an integral domain,
the $t$-adic order $\operatorname{ord}_t$ on $D[t]$ is additive:
$\operatorname{ord}_t(g h) = \operatorname{ord}_t(g) + \operatorname{ord}_t(h)$.
Setting $a_j := \min\left(\operatorname{ord}_t(\overline{X_j}(t)), \, N + 1\right) \ge 1$ for $1 \le j \le n$,
the congruence above is equivalent to $a_1 + \dots + a_n \ge N + 1$.
By definition of $a_j$,
we have $\overline{X_j^{(i)}} = 0$ in $D$ (that is, $X_j^{(i)} \in \mathfrak{q}$) for all $1 \le i \le a_j - 1$.
Decreasing the integers $a_j \ge 1$ if necessary so that $a_1 + \dots + a_n = N + 1$,
we see that $\mathfrak{q}$ contains the monomial prime ideal
\begin{equation*}
  \mathfrak{p}_{a_1, \dots, a_n} := \left(X_j^{(i)} \;\middle|\; 1 \le j \le n, \, 1 \le i \le a_j - 1\right) \subseteq P_{\le N}.
\end{equation*}
Conversely,
modulo $\mathfrak{p}_{a_1, \dots, a_n}$,
each $\overline{X_j}(t)$ is divisible by $t^{a_j}$,
so $\overline{X_1}(t) \cdots \overline{X_n}(t)$ is divisible by $t^{a_1 + \dots + a_n} = t^{N + 1}$ and hence $I \subseteq \mathfrak{p}_{a_1, \dots, a_n}$.
Thus the minimal prime ideals of $I = (u_n, \dots, u_N)$ in $P_{\le N}$ are precisely the prime ideals $\mathfrak{p}_{a_1, \dots, a_n}$ with $a_1 + \dots + a_n = N + 1$ ($a_j \ge 1$).
Since each $\mathfrak{p}_{a_1, \dots, a_n}$ is generated by
\begin{equation*}
  \sum_{j = 1}^n (a_j - 1) = (N + 1) - n = N - n + 1
\end{equation*}
distinct variables of the polynomial ring $P_{\le N}$,
every minimal prime of $I$ has height $N - n + 1$,
so $\operatorname{ht}(u_n, \dots, u_N) = N - n + 1$.
Because $u_n, \dots, u_N$ are $N - n + 1$ homogeneous polynomials of positive weight in the graded polynomial ring $P_{\le N}$ generating an ideal of height $N - n + 1$,
they form a regular sequence in $P_{\le N}$.

(2)–(4) By Part~(1),
the regular sequence hypothesis of Theorem~\ref{thm:general-regular-germs} is satisfied for $F(X_1, \dots, X_n) = X_1 \cdots X_n$ with $d = n$.
Claims (2), (3), and (4) are therefore immediate specializations of parts (1), (2), and (3) of Theorem~\ref{thm:general-regular-germs},
respectively.
\end{proof}

\begin{corollary}[Hilbert--Poincaré Series of $(XY)^e = 0$]
\label{cor:proof-xye-notes}
Let $e \ge 1$ be an integer.
For the algebra $A_e = k[X, Y] / ((XY)^e)$,
we have
\begin{equation}
  \operatorname{HP}_{J_\infty^0(A_e)}(t) = \frac{\mathbb{H}(t)}{1 - t} AG_{e, 1}(t).
\end{equation}
In particular,
for $e = 1$,
since $AG_{1, 1}(t) = 1$,
we recover
\begin{equation}
  \operatorname{HP}_{J_\infty^0(k[X, Y] / (XY))}(t) = \frac{\mathbb{H}(t)}{1 - t};
\end{equation}
for $e = 2$,
we obtain $\frac{\mathbb{H}(t)}{1 - t} AG_{2, 1}(t)$ where $AG_{2, 1}(t) = \sum_{\ell = 0}^\infty \frac{t^{\ell^2 + \ell}}{(t)_\ell}$,
proving Conjecture~\ref{conj:xy2-hilbert-series};
and for every $e \ge 2$,
this proves Conjecture~\ref{conj:xye-generalization}.
\end{corollary}

\begin{proof}
Setting $n = 2$ in Theorem~\ref{thm:normal-crossings-notes} gives $(t)_{n - 1} = (t)_1 = 1 - t$,
and the sum in~\eqref{eq:HP-normal-crossings} ranges over partitions $\mu$ satisfying $\mu_i - \mu_{i + e - 1} \ge 2$ and $\mu_{\mathrm{min}} \ge 2$ (no parts equal to $1$).
By Gordon's combinatorial generalization of the Rogers--Ramanujan identities~\cite{gordon_combinatorial_1961} specialized to $s = 1$
(see also Andrews~\cite{andrews_partitions_1976}, Theorem~7.5),
this generating series is precisely $AG_{e, 1}(t)$ for all $e \ge 1$
(with $AG_{1, 1}(t) = 1$ for $e = 1$).
\end{proof}

\begin{remark}[When Does $(u_d, \dots, u_N)$ Form a Regular Sequence?]
\label{rmk:obstruction-non-squarefree}
Since $P_{\le N}$ is a graded polynomial ring over $k$ and $u_d, \dots, u_N$ are $N - d + 1$ homogeneous polynomials of positive weight,
$(u_d, \dots, u_N)$ is a regular sequence in $P_{\le N}$ if and only if the ideal $(u_d, \dots, u_N)$ has height
\begin{equation*}
  \operatorname{ht}(u_d, \dots, u_N) = N - d + 1
\end{equation*}
(equivalently, every prime ideal $\mathfrak{q} \subseteq P_{\le N}$ containing $(u_d, \dots, u_N)$ satisfies $\operatorname{ht}(\mathfrak{q}) \ge N - d + 1$).
From this criterion we can read off when Proposition~\ref{prop:graded-freeness-finite} applies and when it fails:
\begin{enumerate}
  \item \textbf{Multiplicity bound $d \le n$:}
  For any $F \in k[X_1, \dots, X_n]$ of order $d = \operatorname{ord}_0(F) \ge 1$,
  the prime ideal $\mathfrak{p} := (X_1^{(1)}, \dots, X_n^{(1)}) \subseteq P_{\le N}$ is generated by $n$ variables and therefore has height $\operatorname{ht}(\mathfrak{p}) = n$.
  Modulo $\mathfrak{p}$,
  we have $\operatorname{ord}_t \overline{X_j}(t) \ge 2$ for all $1 \le j \le n$,
  so $\operatorname{ord}_t F(\overline{X_1}(t), \dots, \overline{X_n}(t)) \ge 2 d$ and the $d$ polynomials $u_d, u_{d + 1}, \dots, u_{2d - 1}$ all lie in $\mathfrak{p}$.
  Consequently,
  $\operatorname{ht}(u_d, \dots, u_{2d - 1}) \le \operatorname{ht}(\mathfrak{p}) = n$.
  For $(u_d, \dots, u_{2d - 1})$ to be a regular sequence at $N = 2d - 1$,
  we must therefore have
  \begin{equation*}
    d = \operatorname{ord}_0(F) \le n.
  \end{equation*}
  In particular,
  no hypersurface singularity of multiplicity $d > n$ can satisfy Proposition~\ref{prop:graded-freeness-finite},
  even if $F$ is reduced/squarefree
  (for example, for three distinct lines $F(X, Y) = X Y (X + Y)$ in $k[X, Y]$, we have $d = 3 > n = 2$, and at $N = 5$ the three polynomials $u_3, u_4, u_5$ all lie in the prime ideal $(X^{(1)}, Y^{(1)})$ of height $2 < 3$).

  \item \textbf{Formal invariance and plane curves ($n = 2$):}
  Whether $(u_d, \dots, u_N)$ is a regular sequence depends only on the formal germ of $F = 0$ at the origin:
  a formal coordinate change $\hat{X}_j = \Phi_j(X_1, \dots, X_n)$ with $\Phi(0) = 0$ and invertible Jacobian at $0$ induces a graded algebra automorphism of $P_{\le N}$,
  and multiplying $F$ by a formal unit $U(X) \in k[[X_1, \dots, X_n]]^\times$ ($U(0) = c_0 \ne 0$) replaces $u_l$ by $\tilde{u}_l = c_0 u_l + c_1 u_{l - 1} + \dots + c_{l - d} u_d$,
  which generates the same ideal $(u_d, \dots, u_l)$ for every $l$.
  Consequently,
  Proposition~\ref{prop:graded-freeness-finite} and Theorem~\ref{thm:general-regular-germs} (and in particular Theorem~\ref{thm:normal-crossings-notes}) apply to any $F$ formally isomorphic at $0$ to $X_1 \cdots X_d$ ($d \le n$),
  i.e., any normal crossings divisor with $d \le n$ smooth branches meeting transversely at the origin.
  When $n = 2$,
  a plane curve $F(X, Y) = 0$ satisfies the regularity condition for all $N \ge d$ if $(0, 0)$ is either a smooth point ($d = 1$) or a split node ($d = 2$ whose quadratic part factors into two linearly independent linear forms over $k$, formally isomorphic to $X Y = 0$).
  If instead $d = 2$ with a repeated tangent line,
  regularity fails
  (for instance, for the irreducible cusp $F(X, Y) = Y^2 - X^3$, at $N = 5$ the four polynomials $u_2, u_3, u_4, u_5$ all lie in the height-$3$ prime ideal $\mathfrak{p} = (X^{(1)}, Y^{(1)}, Y^{(2)})$, since $\operatorname{ord}_t \overline{X}(t) \ge 2$ and $\operatorname{ord}_t \overline{Y}(t) \ge 3$ modulo $\mathfrak{p}$ imply $\operatorname{ord}_t(\overline{Y}(t)^2 - \overline{X}(t)^3) \ge 6$, giving $\operatorname{ht}(u_2, u_3, u_4, u_5) \le 3 < 4$).

  \item \textbf{Monomials (squarefree if and only if regular):}
  A monomial $F = X_1^{a_1} \cdots X_n^{a_n}$ ($d = a_1 + \dots + a_n \ge 1$) satisfies the regularity condition for all $N \ge d$ if and only if $F$ is squarefree ($a_j \in \{0, 1\}$ for all $j$).
  Indeed,
  if $F$ is squarefree,
  then up to relabeling variables $F = X_1 \cdots X_d$ with $d \le n$,
  which is covered by Theorem~\ref{thm:normal-crossings-notes}(1).
  Conversely,
  if some exponent satisfies $a_1 \ge 2$,
  then $X_1^{(1)}$ appears to the power $a_1 \ge 2$ in $u_d = [t^d]F(X(t))$ and to the power at least $a_1 - 1 \ge 1$ in every term of $u_{d + 1} = [t^{d + 1}]F(X(t))$;
  hence $X_1^{(1)}$ divides both $u_d$ and $u_{d + 1}$,
  and $(u_d, u_{d + 1})$ fails to be a regular sequence already at $N = d + 1$.
\end{enumerate}

To see explicitly how this non-squarefree obstruction breaks the freeness isomorphism~\eqref{eq:Phi-free-iso} of Proposition~\ref{prop:graded-freeness-finite},
consider $F(X, Y) = X^2 Y$,
which has order $d = 3$.
In truncation $N \ge 4$,
there are $N - 2$ equations $u_3, \dots, u_N$.
In any quotient domain $D = P_{\le N} / \mathfrak{q}$,
elements $\overline{X}(t), \overline{Y}(t) \in D[t]$ with $\alpha = \operatorname{ord}_t(\overline{X}(t)) \ge 1$ and $\beta = \operatorname{ord}_t(\overline{Y}(t)) \ge 1$
satisfy $\overline{X}(t)^2 \overline{Y}(t) \equiv 0 \pmod{t^{N + 1}}$ if and only if $2\alpha + \beta \ge N + 1$.
Taking $\alpha = 2$ and $\beta = N - 3 \ge 1$ shows that $(u_3, \dots, u_N)$ is contained in the monomial prime ideal $\mathfrak{p} := (X^{(1)}, Y^{(1)}, \dots, Y^{(N - 4)}) \subseteq P_{\le N}$,
which is generated by $(2 - 1) + (N - 4) = N - 3$ variables and thus has height $N - 3 < N - 2$.
Hence $\operatorname{ht}(u_3, \dots, u_N) \le N - 3 < N - 2$,
and $(u_3, \dots, u_N)$ is not a regular sequence in $P_{\le N}$.

Explicitly,
$u_3 = (X^{(1)})^2 Y^{(1)}$ (of weight $3$) and $u_4 = X^{(1)}(2X^{(2)}Y^{(1)} + X^{(1)}Y^{(2)})$ (of weight $4$) share the common factor $X^{(1)}Y^{(1)}$ and satisfy the weight-$6$ relation
\begin{equation*}
  g_3 u_3 - g_4 u_4 = 0, \qquad
  g_3 := 2X^{(2)}Y^{(1)} + X^{(1)}Y^{(2)}, \quad
  g_4 := X^{(1)}Y^{(1)}.
\end{equation*}
Here $g_4$ has weight $2$ and $g_3$ has weight $3$.
Since $(u_l \mid l \ge 3)_2 = 0$,
$g_4$ defines a non-zero element of $\left(J_\infty^0(A_F)\right)_2$;
and since $g_3 \notin k \cdot u_3 = (u_l \mid l \ge 3)_3$,
$g_3$ defines a non-zero element of $\left(J_\infty^0(A_F)\right)_3$.
Choosing $g_4$ and $g_3$ among the homogeneous basis lifts $\tilde{e}_\alpha$ of $J_\infty^0(A_F)$ in weights $2$ and $3$,
we see that $V^{(1)} e_{g_3} - V^{(2)} e_{g_4}$ is a non-zero element of $\ker(\Phi)_6$.
Equivalently,
$\dim_k\left(J_\infty^0(A_F)\right)_w$ agrees with $[t^w](\operatorname{HP}_{P_\infty}(t)/\operatorname{HP}_{Q_\infty}(t))$ for $w \le 5$,
but in weight $6$ we have $\dim_k\left(J_\infty^0(A_F)\right)_6 = 48 > 47 = [t^6](\operatorname{HP}_{P_\infty}(t)/\operatorname{HP}_{Q_\infty}(t))$.
Thus the domain of $\Phi$ has weight-$6$ dimension $[t^6]\left(\operatorname{HP}_{Q_\infty}(t)\operatorname{HP}_{J_\infty^0(A_F)}(t)\right) = 66$,
whereas the target $(P_\infty)_6$ has dimension $65$.
Since $\Phi$ is surjective,
$\dim_k \ker(\Phi)_6 = 66 - 65 = 1$,
and Proposition~\ref{prop:graded-freeness-finite} does not apply.
\end{remark}

\subsection{Finite-Variable Recursions and Generalization of BMS Section 5.2}
\label{subsec:finite-variable-recursions}

In Section~5.2 of~\cite{bruschek_arc_2013},
Bruschek, Mourtada, and Schepers gave an alternative commutative-algebraic derivation of the Rogers--Ramanujan recursion for $X^2 = 0$ ($e = 2$) using the infinite-variable polynomial rings $k[W^{(q)} \mid q \ge d]$.
Setting $W^{(q)} := V^{(q - d + 1)}$ (of weight $\operatorname{wt}_d(W^{(q)}) = q$ for $q \ge d$) and truncating the initial ideal~\eqref{eq:shifted-leading-monomials} to the finite-variable ring $k[W^{(d)}, \dots, W^{(N)}]$,
we see both that the auxiliary series $h(d)$ of~\cite{bruschek_arc_2013} is precisely the shifted 1-variable series $\operatorname{HP}_{\left(J_\infty^0(k[V]/(V^2)), \, \operatorname{wt}_d\right)}(t)$ for $(X_1 \cdots X_d)^2 = 0$,
and that the recursion extends in finitely many variables to all $e \ge 2$.

For integers $e \ge 2$, $d \ge 1$, and $N \ge d$,
define the monomial ideal in $k[W^{(d)}, \dots, W^{(N)}]$ (with $\operatorname{wt}(W^{(q)}) = q$):
\begin{equation}
  I_{[d, N]}^{(e)} := \left( (W^{(q)})^{e - j} (W^{(q + 1)})^j \;\middle|\; d \le q \le N, \; 0 \le j \le e - 1 \text{ with } j = 0 \text{ if } q = N \right),
\end{equation}
and for each $1 \le s \le e$,
define the finite-variable quotient algebra
\begin{equation}
  R_{[d, N]}^{(e, s)} := \frac{k[W^{(d)}, \dots, W^{(N)}]}{I_{[d, N]}^{(e, s)}}, \qquad
  I_{[d, N]}^{(e, s)} := I_{[d, N]}^{(e)} + \left((W^{(d)})^s\right),
\end{equation}
with Hilbert--Poincaré polynomial $h_N^{(e, s)}(d) := \operatorname{HP}_{R_{[d, N]}^{(e, s)}}(t) \in \mathbb{N}[t]$
(setting $R_{[d, N]}^{(e, s)} := k$, $I_{[d, N]}^{(e, s)} := (0)$, and $h_N^{(e, s)}(d) := 1$ whenever $d > N$).
Note that for $s = e$,
since $(W^{(d)})^e \in I_{[d, N]}^{(e)}$,
we have $I_{[d, N]}^{(e, e)} = I_{[d, N]}^{(e)}$.
Moreover,
because $I_{[n, N]}^{(e)}$ coincides with the leading ideal of $I_\infty^0(V^e)$ under $W^{(q)} = V^{(q - n + 1)}$ in $\operatorname{wt}_n$-weights $\le N$ by~\eqref{eq:shifted-leading-monomials},
we have
\begin{equation}
  \tau_{\le N} \operatorname{HP}_{\left(J_\infty^0(k[V]/(V^e)), \, \operatorname{wt}_n\right)}(t) = \tau_{\le N} h_N^{(e, e)}(n), \qquad
  \operatorname{HP}_{\left(J_\infty^0(k[V]/(V^e)), \, \operatorname{wt}_n\right)}(t) = \lim_{N \to \infty} h_N^{(e, e)}(n).
\end{equation}

\begin{proposition}[Finite-Variable Ideal Quotients and Andrews--Gordon Recursion]
\label{prop:finite-var-ag-recursion}
Let $e \ge 2$ and $1 \le d \le N$.
\begin{enumerate}
  \item For $s = 1$,
  $I_{[d, N]}^{(e, 1)} = \left(W^{(d)}, \, I_{[d + 1, N]}^{(e, e)}\right)$,
  so $R_{[d, N]}^{(e, 1)} \cong R_{[d + 1, N]}^{(e, e)}$ and
  \begin{equation}
    \label{eq:hN-s1-shift}
    h_N^{(e, 1)}(d) = h_N^{(e, e)}(d + 1).
  \end{equation}
  \item For $2 \le s \le e$,
  we have the ideal identities in $k[W^{(d)}, \dots, W^{(N)}]$:
  \begin{align}
    \left(I_{[d, N]}^{(e, s)}, \, (W^{(d)})^{s - 1}\right) &= I_{[d, N]}^{(e, s - 1)}, \label{eq:ideal-sum-s} \\
    \left(I_{[d, N]}^{(e, s)} : (W^{(d)})^{s - 1}\right) &= \left(W^{(d)}, \, I_{[d + 1, N]}^{(e, e - s + 1)}\right). \label{eq:ideal-colon-s}
  \end{align}
  \item Consequently,
  for all $2 \le s \le e$ and $1 \le d \le N$,
  \begin{equation}
    \label{eq:hN-ag-step}
    h_N^{(e, s)}(d) = h_N^{(e, s - 1)}(d) + t^{(s - 1)d} h_N^{(e, e - s + 1)}(d + 1),
  \end{equation}
  and unrolling~\eqref{eq:hN-ag-step} down to $s = 1$ via~\eqref{eq:hN-s1-shift} gives the finite-variable system
  \begin{equation}
    \label{eq:hN-ag-unrolled}
    h_N^{(e, s)}(d) = \sum_{k = 0}^{s - 1} t^{k d} h_N^{(e, e - k)}(d + 1) \qquad (1 \le s \le e, \; 1 \le d \le N),
  \end{equation}
  with terminal boundary conditions $h_N^{(e, s)}(N + 1) = 1$ for all $1 \le s \le e$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) Setting $W^{(d)} = 0$ eliminates all generators of $I_{[d, N]}^{(e)}$ involving $W^{(d)}$ and leaves $I_{[d + 1, N]}^{(e)} = I_{[d + 1, N]}^{(e, e)}$ in $k[W^{(d + 1)}, \dots, W^{(N)}]$ (or $(0)$ when $d = N$).

(2) Identity~\eqref{eq:ideal-sum-s} is immediate since $(W^{(d)})^e \in ((W^{(d)})^{s - 1})$.
For~\eqref{eq:ideal-colon-s},
if $d = N$,
then $I_{[N, N]}^{(e, s)} = ((W^{(N)})^s)$ in $k[W^{(N)}]$,
so $(I_{[N, N]}^{(e, s)} : (W^{(N)})^{s - 1}) = (W^{(N)})$,
which matches the right-hand side since $I_{[N + 1, N]}^{(e, e - s + 1)} = (0)$ by convention.
For $d < N$,
since $I_{[d, N]}^{(e, s)}$ is a monomial ideal,
its colon ideal with respect to $(W^{(d)})^{s - 1}$ is generated by dividing each monomial generator by its greatest common divisor with $(W^{(d)})^{s - 1}$:
\begin{itemize}
  \item dividing $(W^{(d)})^s$ by $(W^{(d)})^{s - 1}$ gives $W^{(d)}$;
  \item dividing $(W^{(d)})^{e - j}(W^{(d + 1)})^j$ by $(W^{(d)})^{s - 1}$ for $j \ge e - s + 1$ yields $(W^{(d + 1)})^j$
  (whose smallest power is $(W^{(d + 1)})^{e - s + 1}$ at $j = e - s + 1$),
  while terms with $j \le e - s$ retain a factor of $W^{(d)}$ and lie in $(W^{(d)})$;
  \item generators supported on $W^{(d + 1)}, \dots, W^{(N)}$ are coprime to $W^{(d)}$ and generate $I_{[d + 1, N]}^{(e)}$.
\end{itemize}
Thus the colon ideal is $\left(W^{(d)}, \, I_{[d + 1, N]}^{(e)} + ((W^{(d + 1)})^{e - s + 1})\right) = \left(W^{(d)}, \, I_{[d + 1, N]}^{(e, e - s + 1)}\right)$.

(3) Applying the exact sequence for multiplication by $f = (W^{(d)})^{s - 1}$ (of weight $(s - 1)d$),
\begin{equation*}
  \operatorname{HP}_{k[W]/I}(t) = \operatorname{HP}_{k[W]/(I, f)}(t) + t^{\operatorname{wt}(f)} \operatorname{HP}_{k[W]/(I : f)}(t),
\end{equation*}
to~\eqref{eq:ideal-sum-s} and~\eqref{eq:ideal-colon-s} yields~\eqref{eq:hN-ag-step},
and induction on $s$ gives~\eqref{eq:hN-ag-unrolled}.
\end{proof}

\begin{remark}[Algebraic Shift for $(XY)^e = 0$ and Andrews' $q$-Difference Equations]
\label{rmk:algebraic-shift-s1}
For $d = 1$,
the isomorphism $R_{[1, N]}^{(e, 1)} \cong R_{[2, N]}^{(e, e)}$ of Proposition~\ref{prop:finite-var-ag-recursion}(1)
identifies the truncated leading-ideal quotient $R_{[2, N]}^{(e, e)}$ of $(J_\infty^0(k[V]/(V^e)), \operatorname{wt}_2)$
with the $s = 1$ member $R_{[1, N]}^{(e, 1)}$ of the 1-variable family $R_{[1, N]}^{(e, s)}$.
Algebraically,
quotienting the 1-variable fat-point initial ideal $I_{[1, N]}^{(e)}$ by $W^{(1)}$ ($s = 1$) produces the exact initial ideal $I_{[2, N]}^{(e)}$ of the weight-shifted $n = 2$ fiber,
so $\lim_{N \to \infty} h_N^{(e, e)}(2) = \lim_{N \to \infty} h_N^{(e, 1)}(1)$.

Furthermore,
in the limit $N \to \infty$,
the recursion~\eqref{eq:hN-ag-step} for $h_\infty^{(e, s)}(d) := \lim_{N \to \infty} h_N^{(e, s)}(d)$ coincides with Andrews' $q$-difference equations for the series $J_{k, i}(0; x; q)$
(see Lemma~7.2 in Chapter~7 of~\cite{andrews_partitions_1976}, with $J_{k, 0}(0; x; q) = 0$)
under the specialization $k = e$, $i = s$, $q = t$, and $x = t^{d - 1}$:
\begin{equation*}
  J_{e, s}(0; x; t) - J_{e, s - 1}(0; x; t) = (x t)^{s - 1} J_{e, e - s + 1}(0; x t; t).
\end{equation*}
In $J_{e, s}(0; x; t)$,
the exponent of $x$ records the number of parts of a partition,
so setting $x = t^{d - 1}$ adds $d - 1$ to every part and shifts the minimum part from $\ge 1$ to $\ge d$.
Because $h_\infty^{(e, s)}(d) = 1 + O(t^d)$ as $d \to \infty$,
the system~\eqref{eq:hN-ag-unrolled} together with this boundary condition uniquely determines $h_\infty^{(e, s)}(d) = J_{e, s}(0; t^{d - 1}; t)$.
In particular,
Theorem~7.8 of~\cite{andrews_partitions_1976} gives $h_\infty^{(e, e)}(2) = h_\infty^{(e, 1)}(1) = J_{e, 1}(0; 1; t) = AG_{e, 1}(t)$,
providing an analytic proof of Corollary~\ref{cor:proof-xye-notes} that does not rely on Gordon's combinatorial theorem.
\end{remark}

\begin{corollary}[Shifted Rogers--Ramanujan Recursion for $e = 2$]
\label{cor:shifted-rr-recursion}
For $e = 2$,
write $h_N(d) := h_N^{(2, 2)}(d)$.
Then for all $1 \le d \le N$,
\begin{equation}
  \label{eq:bms-finite-recursion}
  h_N(d) = h_N(d + 1) + t^d h_N(d + 2), \qquad h_N(N + 1) = h_N(N + 2) = 1.
\end{equation}
For any fixed $n \ge 1$ and any $n \le d \le N + 1$,
\begin{equation}
  \label{eq:hN-Ad-Bd}
  h_N(n) = \alpha_d^{(n)}(t) \, h_N(d) + \beta_{d + 1}^{(n)}(t) \, h_N(d + 1),
\end{equation}
where the polynomials $\alpha_d^{(n)}(t), \beta_d^{(n)}(t) \in \mathbb{N}[t]$ satisfy
\begin{equation*}
  \alpha_d^{(n)}(t) = \alpha_{d - 1}^{(n)}(t) + \beta_d^{(n)}(t), \qquad
  \beta_{d + 1}^{(n)}(t) = t^{d - 1} \alpha_{d - 1}^{(n)}(t) \qquad (d \ge n + 1),
\end{equation*}
with initial conditions $\alpha_n^{(n)}(t) = \alpha_{n + 1}^{(n)}(t) = 1$ and $\beta_{n + 1}^{(n)}(t) = 0$.
Evaluating~\eqref{eq:hN-Ad-Bd} at $d = N + 1$ yields the exact polynomial identity $S_N^{(n)}(t) := h_N(n) = \alpha_{N + 2}^{(n)}(t)$,
so the finite-variable Hilbert--Poincaré polynomials $S_N^{(n)}(t)$ themselves satisfy the $3$-term Schur / Andrews--Baxter recursion:
\begin{equation}
  \label{eq:schur-3term-r}
  S_N^{(n)}(t) = S_{N - 1}^{(n)}(t) + t^N S_{N - 2}^{(n)}(t) \qquad (N \ge n), \qquad
  S_{n - 2}^{(n)}(t) = S_{n - 1}^{(n)}(t) = 1.
\end{equation}
\end{corollary}

\begin{proof}
Setting $e = 2$ and $s = 2$ in~\eqref{eq:hN-ag-unrolled} gives $h_N(d) = h_N(d + 1) + t^d h_N^{(2, 1)}(d + 1) = h_N(d + 1) + t^d h_N(d + 2)$ by~\eqref{eq:hN-s1-shift}.
Relation~\eqref{eq:hN-Ad-Bd} follows by induction on $d$ for $n \le d \le N + 1$ by substituting $h_N(d) = h_N(d + 1) + t^d h_N(d + 2)$.
Finally,
setting $d = N + 1$ in~\eqref{eq:hN-Ad-Bd} and using $h_N(N + 1) = h_N(N + 2) = 1$ gives
\begin{equation*}
  h_N(n) = \alpha_{N + 1}^{(n)}(t) + \beta_{N + 2}^{(n)}(t) = \alpha_{N + 2}^{(n)}(t),
\end{equation*}
which satisfies~\eqref{eq:schur-3term-r} for all $N \ge n$ with $S_{n - 2}^{(n)}(t) = \alpha_n^{(n)}(t) = 1$ and $S_{n - 1}^{(n)}(t) = \alpha_{n + 1}^{(n)}(t) = 1$.
\end{proof}

\section{Graded Vector Spaces, Modules, and Tensor Products}
\label{app:graded-tensor-products}

Let $k$ be a field.
An \defterm{$\mathbb{N}$-graded $k$-vector space} is a $k$-vector space $V$ equipped with a direct sum decomposition
\begin{equation*}
  V = \bigoplus_{w \in \mathbb{N}} V_w
\end{equation*}
into $k$-subspaces $V_w$
(with the convention that $V_w = 0$ for integers $w < 0$).
An element $v \in V_w$ is said to be \defterm{homogeneous} of \defterm{weight} $w$,
written $\operatorname{wt}(v) = w$.
When each homogeneous component $V_w$ is finite-dimensional over $k$,
the \defterm{Hilbert--Poincaré series of $V$} is the formal power series
\begin{equation*}
  \operatorname{HP}_V(t) = \sum_{w \in \mathbb{N}} \dim_k(V_w) t^w \in \mathbb{N}[[t]].
\end{equation*}

More generally,
let $R = \bigoplus_{w \in \mathbb{N}} R_w$ be an $\mathbb{N}$-graded $k$-algebra.
An \defterm{$\mathbb{N}$-graded $R$-module} is an $R$-module $M$ equipped with an $\mathbb{N}$-graded $k$-vector space decomposition $M = \bigoplus_{w \in \mathbb{N}} M_w$ such that $R_i M_j \subseteq M_{i + j}$ for all $i, j \in \mathbb{N}$.
An $R$-linear map $\varphi \colon M \to N$ between $\mathbb{N}$-graded $R$-modules is a \defterm{graded homomorphism} (of weight $0$) if $\varphi(M_w) \subseteq N_w$ for all $w \in \mathbb{N}$.

Given an $\mathbb{N}$-graded $R$-module (or $k$-vector space) $M$ and an integer $s \in \mathbb{N}$,
the \defterm{shifted graded module} $M(-s)$ is the same underlying $R$-module $M$ endowed with the shifted grading
\begin{equation}
  \label{eq:app-shifted-grading}
  M(-s)_w := M_{w - s} \qquad (w \in \mathbb{N}),
\end{equation}
where $M_{w - s} = 0$ for $w < s$.
Thus a homogeneous element $m \in M_j$ has weight $w = j + s$ when viewed as an element of $M(-s)$.
If each $M_w$ is finite-dimensional over $k$,
re-indexing the sum gives
\begin{equation}
  \label{eq:app-hp-shift}
  \operatorname{HP}_{M(-s)}(t) = \sum_{w = s}^\infty \dim_k(M_{w - s}) t^w = t^s \operatorname{HP}_M(t).
\end{equation}
In particular,
viewing the base field $k$ as an $\mathbb{N}$-graded $k$-vector space concentrated in weight $0$
(so $k_0 = k$ and $k_w = 0$ for $w \ge 1$),
the shifted space $k(-s)$ is a $1$-dimensional $k$-vector space concentrated in weight $s$,
with $\operatorname{HP}_{k(-s)}(t) = t^s$.

Now let $A = \bigoplus_{i \in \mathbb{N}} A_i$ and $B = \bigoplus_{j \in \mathbb{N}} B_j$ be $\mathbb{N}$-graded $k$-vector spaces.
Because the tensor product over $k$ distributes over direct sums,
we have a canonical decomposition
\begin{equation*}
  A \otimes_k B
  = \left(\bigoplus_{i \in \mathbb{N}} A_i\right) \otimes_k \left(\bigoplus_{j \in \mathbb{N}} B_j\right)
  \cong \bigoplus_{w \in \mathbb{N}} \left(\bigoplus_{i + j = w} A_i \otimes_k B_j\right).
\end{equation*}
The \defterm{tensor product grading} on $A \otimes_k B$ is defined by setting its weight-$w$ homogeneous component to be
\begin{equation}
  \label{eq:app-tensor-grading}
  (A \otimes_k B)_w := \bigoplus_{i + j = w} A_i \otimes_k B_j \qquad (w \in \mathbb{N}).
\end{equation}
On homogeneous pure tensors $a \in A_i$ and $b \in B_j$,
the weights add:
\begin{equation*}
  \operatorname{wt}(a \otimes b) = \operatorname{wt}(a) + \operatorname{wt}(b) = i + j.
\end{equation*}
When each $A_i$ and $B_j$ is finite-dimensional over $k$,
taking dimensions in~\eqref{eq:app-tensor-grading} yields the Cauchy product formula
\begin{equation*}
  \dim_k\left((A \otimes_k B)_w\right) = \sum_{i + j = w} \dim_k(A_i) \dim_k(B_j),
\end{equation*}
so the Hilbert--Poincaré series of $A \otimes_k B$ factors as the product
\begin{equation}
  \label{eq:app-hp-tensor-product}
  \operatorname{HP}_{A \otimes_k B}(t) = \operatorname{HP}_A(t) \cdot \operatorname{HP}_B(t).
\end{equation}

Suppose further that $M$ is an $\mathbb{N}$-graded $R$-module and $V$ is an $\mathbb{N}$-graded $k$-vector space.
Then $V \otimes_k M$ is an $\mathbb{N}$-graded $R$-module with the grading~\eqref{eq:app-tensor-grading} and the $R$-action on the right tensor factor,
$r \cdot (v \otimes m) := v \otimes (r m)$.
Choosing any homogeneous $k$-vector space basis $(e_\alpha)_{\alpha \in \mathcal{I}}$ of $V$ with $w_\alpha = \operatorname{wt}(e_\alpha)$ decomposes $V$ into $1$-dimensional shifted subspaces:
\begin{equation*}
  V = \bigoplus_{\alpha \in \mathcal{I}} k \cdot e_\alpha \cong \bigoplus_{\alpha \in \mathcal{I}} k(-w_\alpha).
\end{equation*}
For each $\alpha \in \mathcal{I}$ and $w \in \mathbb{N}$,
\begin{equation*}
  \left(k(-w_\alpha) \otimes_k M\right)_w
  = \bigoplus_{i + j = w} k(-w_\alpha)_i \otimes_k M_j
  = k(-w_\alpha)_{w_\alpha} \otimes_k M_{w - w_\alpha}
  \cong M_{w - w_\alpha}
  = M(-w_\alpha)_w.
\end{equation*}
Summing over $\alpha \in \mathcal{I}$ gives a canonical isomorphism of $\mathbb{N}$-graded $R$-modules
\begin{equation}
  \label{eq:app-shifted-sum-tensor}
  \bigoplus_{\alpha \in \mathcal{I}} M(-w_\alpha) \xrightarrow{\;\sim\;} V \otimes_k M, \qquad
  (m_\alpha)_{\alpha \in \mathcal{I}} \longmapsto \sum_{\alpha \in \mathcal{I}} e_\alpha \otimes m_\alpha.
\end{equation}
Finally,
for any $\mathbb{N}$-graded $R$-module $N$ and any shift $s \in \mathbb{N}$,
the canonical $R$-module isomorphism $R \otimes_R N \cong N$ respects the shifted grading~\eqref{eq:app-shifted-grading} and gives an isomorphism of $\mathbb{N}$-graded $R$-modules
\begin{equation}
  \label{eq:app-shifted-base-change}
  R(-s) \otimes_R N \xrightarrow{\;\sim\;} N(-s), \qquad
  r \otimes n \longmapsto r n,
\end{equation}
since for $r \in R_i = R(-s)_{i + s}$ and $n \in N_j$,
the product $r n \in N_{i + j} = N(-s)_{i + j + s}$ has weight $(i + s) + j$.
